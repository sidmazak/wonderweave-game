   107→  const doneGoalsRef = React.useRef<Set<string>>(new Set())
   108→
   109→  const [busy, setBusy] = React.useState(true)
   110→  const busyRef = React.useRef(true)
   111→  const lock = React.useCallback((v: boolean) => {

   115→
   116→  const [selected, setSelected] = React.useState<Cell | null>(null)

   123→  const hintTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)

   128→  const [goalFlash, setGoalFlash] = React.useState<{ id: number } | null>(null)
   129→  const [toast, setToast] = React.useState<string | null>(null)
   130→
   131→  /* ------- board shake for big weaves ------- */
   132→  const [shakeId, setShakeId] = React.useState(0)
   133→  const shakeTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   134→  const triggerShake = React.useCallback(() => {
   135→    const id = Date.now()

   137→    if (shakeTimer.current) clearTimeout(shakeTimer.current)

   139→      shakeTimer.current = null
   140→      setShakeId((cur) => (cur === id ? 0 : cur))

   142→  }, [])
   143→  React.useEffect(() => {
   144→    return () => {
   145→      if (shakeTimer.current) clearTimeout(shakeTimer.current)

   147→  }, [])
   148→
   149→  /* ------- low-moves heartbeat (3 / 2 / 1) ------- */
   150→  const urgentRef = React.useRef<number | null>(null)
   151→
   152→  const [status, setStatus] = React.useState<'playing' | 'won' | 'lost'>('playing')
   153→  const statusRef = React.useRef<'playing' | 'won' | 'lost'>('playing')
   154→  const [result, setResult] = React.useState<LevelResult | null>(null)
   155→  const [rewards, setRewards] = React.useState<RewardSummary | null>(null)
   156→  const [paused, setPaused] = React.useState(false)
   157→  const pausedRef = React.useRef(false)
   158→  const [armedBooster, setArmedBooster] = React.useState<BoosterKind | null>(null)
   159→  const armedRef = React.useRef<BoosterKind | null>(null)
   160→  const setArmed = React.useCallback((k: BoosterKind | null) => {
   161→    armedRef.current = k
   162→    setArmedBooster(k)
   163→  }, [])
   164→
   165→  const wrapRef = React.useRef<HTMLDivElement>(null)
   166→  const boardRef = React.useRef<HTMLDivElement>(null)
   167→  const [size, setSize] = React.useState({ w: 0, h: 0 })
   168→
   169→  /* ------- board sizing ------- */
   170→  React.useEffect(() => {
   171→    const el = wrapRef.current
   172→    if (!el) return
   173→    const measure = () => {
   174→      const cs = getComputedStyle(el)
   175→      const availW = el.clientWidth - parseFloat(cs.paddingLeft || '0') - parseFloat(cs.paddingRight || '0')
   176→      const availH = el.clientHeight - parseFloat(cs.paddingTop || '0') - parseFloat(cs.paddingBottom || '0')
   177→      // frame adds 2*3px border + 2*5px padding = 16px; keep a 4px slack
   178→      const target = Math.max(120, Math.min(availW - 20, (availH - 20) * (level.cols / level.rows)))
   179→      setSize({ w: target, h: (target * level.rows) / level.cols })
   180→    }
   181→    measure()
   182→    const ro = new ResizeObserver(measure)
   183→    ro.observe(el)
   184→    return () => ro.disconnect()
   185→  }, [level.rows, level.cols])
   186→
   187→  /* ------- deal animation ------- */
   188→  const clearSpawnedNextFrame = React.useCallback(() => {
   189→    requestAnimationFrame(() =>
   190→      requestAnimationFrame(() => {
   191→        const g = cloneGrid(gridRef.current)
   192→        let dirty = false
   193→        for (const row of g) {
   194→          for (const t of row) {
   195→            if (t && t.spawned) {
   196→              delete t.spawned
   197→              dirty = true
   198→            }
   199→          }
   200→        }
   201→        if (dirty) setGridBoth(g)
   202→      }),
   203→    )
   204→  }, [setGridBoth])
   205→
   206→  React.useEffect(() => {
   207→    initAudio()
   208→    discover('entity:lantern')
   209→    // lore follows the 12 core themes so echo chapters still unlock them

   211→    clearSpawnedNextFrame()
   212→    const t = setTimeout(() => lock(false), 480 + level.cols * 26)
   213→    return () => clearTimeout(t)
   214→  }, [clearSpawnedNextFrame, lock, level.cols])
   215→
   216→  const resetLevel = React.useCallback(() => {
   217→    const g = createGrid(level.rows, level.cols, level.types)
   218→    for (const row of g) for (const t of row) if (t) t.spawned = true
   219→    setGridBoth(g)
   220→    clearSpawnedNextFrame()
   221→    scoreRef.current = 0
   222→    setScore(0)
   223→    movesRef.current = level.moves
   224→    setMovesLeft(level.moves)
   225→    collectedRef.current = {}
   226→    setCollected({})
   227→    doneGoalsRef.current = new Set()
   228→    setSelectedBoth(null)
   229→    setArmed(null)
   230→    setHint(null)
   231→    setCombo(null)
   232→    urgentRef.current = null
   233→    setFloaters([])
   234→    setResult(null)
   235→    setRewards(null)
   236→    setStatus('playing')
   237→    statusRef.current = 'playing'
   238→    setPaused(false)
   239→    pausedRef.current = false
   240→    lock(true)
   241→    setTimeout(() => lock(false), 480 + level.cols * 26)
   242→  }, [clearSpawnedNextFrame, level, lock, setArmed, setGridBoth, setSelectedBoth])
   243→
   244→  /* ------- helpers ------- */
   245→  const showToast = React.useCallback((msg: string) => {
   246→    setToast(msg)
   247→    setTimeout(() => setToast((cur) => (cur === msg ? null : cur)), 2200)
   248→  }, [])
   249→
   250→  const addFloater = React.useCallback((r: number, c: number, text: string) => {
   251→    const id = floaterSeq++
   252→    setFloaters((f) => [...f.slice(-6), { id, r, c, text }])
   253→    setTimeout(() => setFloaters((f) => f.filter((x) => x.id !== id)), 1000)
   254→  }, [])
   255→
   256→  const showCombo = React.useCallback((n: number, text?: string) => {
   257→    const id = comboSeq++
   258→    setCombo({ id, n, text })
   259→    setTimeout(() => setCombo((cur) => (cur?.id === id ? null : cur)), 1400)
   260→  }, [])
   261→
   262→  const finishLevel = React.useCallback(
   263→    (won: boolean) => {
   264→      const bonus = won ? movesRef.current * 100 : 0
   265→      const finalScore = scoreRef.current + bonus
   266→      const stars = won ? (finalScore >= level.star3 ? 3 : finalScore >= level.star2 ? 2 : 1) : 0
   267→      const res: LevelResult = { levelId: level.id, won, score: finalScore, stars, movesLeft: movesRef.current }
   268→      statusRef.current = won ? 'won' : 'lost'
   269→      setStatus(won ? 'won' : 'lost')
   270→      setResult(res)
   271→      if (won) {
   272→        sfx.win()
   273→        for (let i = 0; i < stars; i++) setTimeout(() => sfx.star(i), 500 + i * 380)
   274→        setRewards(onWin(res))
   275→      } else {
   276→        sfx.lose()
   277→      }
   278→    },
   279→    [level, onWin],
   280→  )
   281→
   282→  const checkObjective = React.useCallback((): boolean => {
   283→    const obj = level.objective
   284→    if (obj.kind === 'score') return scoreRef.current >= (obj.score ?? Infinity)
   285→    return (obj.collect ?? []).every((goal) => (collectedRef.current[goal.type] ?? 0) >= goal.count)
   286→  }, [level])
   287→
   288→  /* ------- blast & cascade ------- */
   289→  const blast = React.useCallback(
   290→    async (plan: ClearPlan, cascade: number): Promise<void> => {
   291→      const g = cloneGrid(gridRef.current)
   292→      const cellsArr = [...plan.cells].map((k) => {
   293→        const [r, c] = k.split(',').map(Number)
   294→        return { r, c }
   295→      })
   296→      if (cellsArr.length === 0 && plan.promotions.size === 0) return
   297→
   298→      // big weaves rattle the frame
   299→      if (cellsArr.length >= 10) triggerShake()
   300→
   301→      const mult = cascade + 1
   302→      const gained = cellsArr.length * 20 * mult + plan.promotions.size * 60
   303→      scoreRef.current += gained
   304→      setScore(scoreRef.current)
   305→
   306→      // collection + codex discovery
   307→      const col = { ...collectedRef.current }
   308→      let lineHit = false
   309→      let bombHit = false
   310→      const seenTypes = new Set<string>()
   311→      for (const { r, c } of cellsArr) {
   312→        const t = g[r]?.[c]
   313→        if (!t) continue
   314→        col[t.type] = (col[t.type] ?? 0) + 1
   315→        seenTypes.add(t.type)
   316→        if (t.special === 'lineH' || t.special === 'lineV') lineHit = true
   317→        if (t.special === 'bomb') bombHit = true
   318→      }
   319→      collectedRef.current = col
   320→      setCollected(col)
   321→      if (seenTypes.size > 0) {
   322→        const lumenIds = [...seenTypes].map((t) => LUMEN_ID_BY_TYPE[t]).filter(Boolean)
   323→        if (lumenIds.length > 0) discover(...lumenIds)
   324→      }
   325→      for (const promo of plan.promotions.values()) {
   326→        if (promo.special === 'prism') discover('lumen:vitriol')
   327→      }
   328→      if (lineHit) sfx.line()
   329→      if (bombHit) sfx.bomb()
   330→
   331→      // floater at centroid
   332→      if (cellsArr.length > 0) {
   333→        const ar = cellsArr.reduce((s, x) => s + x.r, 0) / cellsArr.length
   334→        const ac = cellsArr.reduce((s, x) => s + x.c, 0) / cellsArr.length
   335→        addFloater(ar, ac, `+${gained}`)
   336→      }
   337→
   338→      // pop
   339→      for (const k of plan.cells) {
   340→        const [r, c] = k.split(',').map(Number)
   341→        const t = g[r]?.[c]
   342→        if (t) t.clearing = true
   343→      }
   344→      setGridBoth(g)
   345→      sfx.pop(cascade)
   346→      vibrate(12)
   347→      await sleep(250)
   348→
   349→      // clear + gravity
   350→      applyClear(g, plan)
   351→      const grav = applyGravity(g, level.types)
   352→      setGridBoth(g)
   353→      if (grav.spawnedIds.length > 0) {
   354→        await sleep(430 + level.cols * 26 + 60)
   355→        const g2 = cloneGrid(gridRef.current)
   356→        let dirty = false
   357→        for (const row of g2) {
   358→          for (const t of row) {
   359→            if (t && t.spawned) {
   360→              delete t.spawned
   361→              dirty = true
   362→            }
   363→          }
   364→        }
   365→        if (dirty) setGridBoth(g2)
   366→      } else {
   367→        await sleep(300)
   368→      }
   369→
   370→      // combo banner
   371→      if (mult >= 2) showCombo(mult)
   372→
   373→      // goal flash
   374→      if (level.objective.kind === 'collect') {
   375→        for (const goal of level.objective.collect ?? []) {
   376→          const done = (collectedRef.current[goal.type] ?? 0) >= goal.count
   377→          if (done && !doneGoalsRef.current.has(goal.type)) {
   378→            doneGoalsRef.current.add(goal.type)
   379→            setGoalFlash({ id: Date.now() })
   380→            sfx.specialCreate()
   381→            setTimeout(() => setGoalFlash(null), 1400)
   382→          }
   383→        }
   384→      }
   385→    },
   386→    [addFloater, level, setGridBoth, showCombo, triggerShake, vibrate],
   387→  )
   388→
   389→  const cascadeLoop = React.useCallback(
   390→    async (swapped?: Cell[]): Promise<void> => {
   391→      let cascade = 0
   392→      for (;;) {
   393→        const g = cloneGrid(gridRef.current)
   394→        const shapes = findShapes(g, cascade === 0 ? swapped : undefined)
   395→        if (shapes.length === 0) break
   396→        const plan = planClear(g, shapes, cascade === 0 ? swapped : undefined)
   397→        await blast(plan, cascade)
   398→        cascade++
   399→      }
   400→    },
   401→    [blast],
   402→  )
   403→
   404→  const activatePrism = React.useCallback(
   405→    async (a: Cell, b: Cell): Promise<void> => {
   406→      const g = gridRef.current
   407→      const ta = g[a.r]?.[a.c]
   408→      const tb = g[b.r]?.[b.c]
   409→      if (!ta || !tb) return
   410→      const aIsPrism = ta.special === 'prism'
   411→      const other = aIsPrism ? tb : ta
   412→      sfx.prism()
   413→      discover('lumen:vitriol')
   414→      const seed = new Set<string>()
   415→      if (other.special === 'prism') {
   416→        for (let r = 0; r < level.rows; r++) for (let c = 0; c < level.cols; c++) seed.add(key(r, c))
   417→      } else {
   418→        for (let r = 0; r < level.rows; r++) {
   419→          for (let c = 0; c < level.cols; c++) {
   420→            if (g[r][c]?.type === other.type) seed.add(key(r, c))
   421→          }
   422→        }
   423→      }
   424→      const cells = expandSpecials(g, seed)
   425→      await blast({ cells, promotions: new Map() }, 0)
   426→      await cascadeLoop()
   427→    },
   428→    [blast, cascadeLoop, level.rows, level.cols],
   429→  )
   430→
   431→  const afterMove = React.useCallback(async (): Promise<void> => {
   432→    if (checkObjective()) {
   433→      finishLevel(true)
   434→      return
   435→    }
   436→    if (movesRef.current <= 0) {
   437→      finishLevel(false)
   438→      return
   439→    }
   440→    if (!hasAnyMove(cloneGrid(gridRef.current))) {
   441→      showToast('The loom is stuck — reshuffling threads…')
   442→      sfx.shuffle()
   443→      const ng = shuffleGrid(cloneGrid(gridRef.current), level.types)
   444→      setGridBoth(ng)
   445→      await sleep(450)
   446→      await cascadeLoop()
   447→    }
   448→  }, [cascadeLoop, checkObjective, finishLevel, level.types, setGridBoth, showToast])
   449→
   450→  /* ------- swap flow ------- */
   451→  const attemptSwap = React.useCallback(
   452→    async (a: Cell, b: Cell, opts?: { free?: boolean }): Promise<void> => {
   453→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   454→      if (!a || !b || typeof a.r !== 'number' || typeof b.r !== 'number') return
   455→      if (!adjacent(a, b)) return
   456→      lock(true)
   457→      setSelectedBoth(null)
   458→      setHint(null)
   459→
   460→      const g0 = cloneGrid(gridRef.current)
   461→      const ta = g0[a.r]?.[a.c]
   462→      const tb = g0[b.r]?.[b.c]
   463→      if (!ta || !tb) {
   464→        lock(false)
   465→        return
   466→      }
   467→
   468→      const swapped = cloneGrid(g0)
   469→      swapped[a.r][a.c] = { ...tb }
   470→      swapped[b.r][b.c] = { ...ta }
   471→      setGridBoth(swapped)
   472→      sfx.swap()
   473→      await sleep(190)
   474→
   475→      const g1 = gridRef.current
   476→      const prismInvolved = ta.special === 'prism' || tb.special === 'prism'
   477→      // the full special+special weave table — non-null only when BOTH tiles are special
   478→      const combo = comboPlan(g1, a, b)
   479→      // g1 already contains the swapped tiles, so any shape present now was created by the swap
   480→      const valid = prismInvolved || combo !== null || findShapes(g1).length > 0
   481→
   482→      if (!valid) {
   483→        setGridBoth(g0)
   484→        sfx.invalid()
   485→        await sleep(190)
   486→        lock(false)
   487→        return
   488→      }
   489→
   490→      if (!opts?.free) {
   491→        movesRef.current -= 1
   492→        setMovesLeft(movesRef.current)
   493→        // heartbeat as the loom runs out of moves (3 / 2 / 1)
   494→        if (movesRef.current >= 1 && movesRef.current <= 3 && urgentRef.current !== movesRef.current) {
   495→          urgentRef.current = movesRef.current
   496→          sfx.urgent()
   497→        }
   498→      }
   499→      vibrate(10)
   500→
   501→      if (combo) {
   502→        // special + special weave: tiered fanfare, banner, bonus, then the blast
   503→        const tier = combo.kind === 'blackhole' ? 3 : combo.kind === 'lineStorm' || combo.kind === 'bombStorm' ? 2 : 1
   504→        sfx.combo(tier)
   505→        vibrate(24)
   506→        showCombo(1, comboLabel(combo.kind))
   507→        scoreRef.current += combo.bonus
   508→        setScore(scoreRef.current)
   509→        addFloater((a.r + b.r) / 2, (a.c + b.c) / 2, `+${combo.bonus}`)
   510→        triggerShake()
   511→        await blast({ cells: expandSpecials(g1, combo.cells), promotions: new Map() }, 0)
   512→        await cascadeLoop()
   513→      } else if (prismInvolved) {
   514→        await activatePrism(a, b)
   515→      } else {
   516→        await cascadeLoop([a, b])
   517→      }
   518→      await afterMove()
   519→      lock(false)
   520→    },
   521→    [
   522→      activatePrism,
   523→      addFloater,
   524→      afterMove,
   525→      blast,
   526→      cascadeLoop,
   527→      lock,
   528→      setGridBoth,
   529→      setSelectedBoth,
   530→      showCombo,
   531→      triggerShake,
   532→      vibrate,
   533→    ],
   534→  )
   535→
   536→  /* ------- boosters ------- */
   537→  const armBooster = React.useCallback(
   538→    (kind: BoosterKind) => {
   539→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   540→      if (inventory[kind] <= 0) return
   541→      initAudio()
   542→      if (kind === 'lens') {
   543→        // LENS executes the weave instantly (no move spent)
   544→        if (!spendBooster('lens')) return
   545→        setArmed(null)
   546→        setSelectedBoth(null)
   547→        sfx.lens()
   548→        const moves = findAllMoves(cloneGrid(gridRef.current))
   549→        if (moves.length === 0) {
   550→          showToast('No threads available — the loom reshuffles…')
   551→          const ng = shuffleGrid(cloneGrid(gridRef.current), level.types)
   552→          setGridBoth(ng)
   553→          return
   554→        }
   555→        void attemptSwap(moves[0][0], moves[0][1], { free: true })
   556→        return
   557→      }
   558→      sfx.booster()
   559→      if (armedRef.current === 'null') {
   560→        setArmed(null)
   561→      } else {
   562→        setArmed('null')
   563→        showToast('Tap any charm to unweave it…')
   564→      }
   565→      setSelectedBoth(null)
   566→    },
   567→    [attemptSwap, inventory, level.types, setArmed, setGridBoth, setSelectedBoth, showToast, spendBooster],
   568→  )
   569→
   570→  const nullifyCell = React.useCallback(
   571→    async (cell: Cell): Promise<void> => {
   572→      if (busyRef.current || statusRef.current !== 'playing') return
   573→      if (!spendBooster('null')) {
   574→        setArmed(null)
   575→        return
   576→      }
   577→      lock(true)
   578→      setArmed(null)
   579→      setSelectedBoth(null)
   580→      setHint(null)
   581→      sfx.nullify()
   582→      vibrate(16)
   583→
   584→      const g = cloneGrid(gridRef.current)
   585→      const t = g[cell.r]?.[cell.c]
   586→      if (!t) {
   587→        lock(false)
   588→        return
   589→      }
   590→      t.clearing = true
   591→      setGridBoth(g)
   592→      await sleep(230)
   593→
   594→      applyClear(g, { cells: new Set([key(cell.r, cell.c)]), promotions: new Map() })
   595→      const grav = applyGravity(g, level.types)
   596→      setGridBoth(g)
   597→      if (grav.spawnedIds.length > 0) await sleep(430 + level.cols * 26 + 60)
   598→      else await sleep(280)
   599→      const g2 = cloneGrid(gridRef.current)
   600→      let dirty = false
   601→      for (const row of g2) {
   602→        for (const tl of row) {
   603→          if (tl && tl.spawned) {
   604→            delete tl.spawned
   605→            dirty = true
   606→          }
   607→        }
   608→      }
   609→      if (dirty) setGridBoth(g2)
   610→
   611→      await cascadeLoop()
   612→      await afterMove()
   613→      lock(false)
   614→    },
   615→    [afterMove, cascadeLoop, level.cols, level.types, lock, setArmed, setGridBoth, setSelectedBoth, spendBooster, vibrate],
   616→  )
   617→
   618→  /* ------- pointer input ------- */
   619→  const cellFromPoint = React.useCallback(
   620→    (x: number, y: number): Cell | null => {
   621→      const el = boardRef.current
   622→      if (!el) return null
   623→      const rect = el.getBoundingClientRect()
   624→      const c = Math.floor(((x - rect.left) / rect.width) * level.cols)
   625→      const r = Math.floor(((y - rect.top) / rect.height) * level.rows)

   629→    [level.cols, level.rows],
   630→  )
   631→

   636→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return

   638→        const moves = findAllMoves(cloneGrid(gridRef.current))
   639→        if (moves.length > 0) setHint(moves[Math.floor(Math.random() * moves.length)])
   640→      } catch {
   641→        /* never let the hint probe crash the game */
   642→      }
   643→    }, 6000)
   644→  }, [])
   645→
   646→  React.useEffect(() => {
   647→    resetHintTimer()
   648→    return () => {
   649→      if (hintTimer.current) clearTimeout(hintTimer.current)
   650→    }
   651→    // re-arm after activity
   652→  }, [score, movesLeft, busy, resetHintTimer])
   653→
   654→  const onPointerDown = (e: React.PointerEvent) => {
   655→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   656→    const cell = cellFromPoint(e.clientX, e.clientY)

   659→    if (armedRef.current === 'null') {
   660→      void nullifyCell(cell)
   661→      return
   662→    }
   663→    dragRef.current = { startCell: cell, x: e.clientX, y: e.clientY, done: false }
   664→  }
   665→  const onPointerMove = (e: React.PointerEvent) => {
   666→    const d = dragRef.current
   667→    if (!d || d.done || busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   668→    const dx = e.clientX - d.x
   669→    const dy = e.clientY - d.y
   670→    const el = boardRef.current
   671→    if (!el) return

   673→    if (Math.hypot(dx, dy) >= cellPx * 0.35) {
   674→      d.done = true
   675→      const dir =
   676→        Math.abs(dx) > Math.abs(dy)
   677→          ? { dr: 0, dc: dx > 0 ? 1 : -1 }
   678→          : { dr: dy > 0 ? 1 : -1, dc: 0 }
   679→      const target = { r: d.startCell.r + dir.dr, c: d.startCell.c + dir.dc }
   680→      if (target.r >= 0 && target.c >= 0 && target.r < level.rows && target.c < level.cols) {
   681→        setSelectedBoth(null)
   682→        void attemptSwap(d.startCell, target)
   683→      }
   684→    }
   685→  }
   686→
   687→  const onPointerUp = (e: React.PointerEvent) => {
   688→    const d = dragRef.current
   689→    dragRef.current = null
   690→    if (!d || d.done) return
   691→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   692→    const cell = cellFromPoint(e.clientX, e.clientY)
   693→    if (!cell) {
   694→      setSelectedBoth(null)
   695→      return
   696→    }
   697→    if (armedRef.current === 'null') {
   698→      void nullifyCell(cell)
   699→      return
   700→    }
   701→    // pure tap/select logic — side effects happen outside any state updater
   702→    const sel = selectedRef.current
   703→    if (!sel) {
   704→      sfx.select()
   705→      setSelectedBoth(cell)
   706→      return
   707→    }
   708→    if (sel.r === cell.r && sel.c === cell.c) {
   709→      setSelectedBoth(null)
   710→      return
   711→    }
   712→    if (adjacent(sel, cell)) {
   713→      const a = sel
   714→      setSelectedBoth(null)
   715→      void attemptSwap(a, cell)
   716→      return
   717→    }
   718→    sfx.select()
   719→    setSelectedBoth(cell)
   720→  }
   721→
   722→  /* ------- render ------- */
   723→  const obj = level.objective
   724→  const bgKey = level.chapter > 0 ? chapterOf(level.id).bg : 'bg-altar'
   725→  const stageLabel = isDaily ? 'DAILY PAGE' : `STAGE ${stageTitle(level)}`

   733→            const c = i % level.cols

   736→                key={i}
   737→                aria-hidden
   738→                className="absolute"
   739→                style={{
   740→                  left: `${(c * 100) / level.cols}%`,
   741→                  top: `${(r * 100) / level.rows}%`,
   742→                  width: `${100 / level.cols}%`,
   743→                  height: `${100 / level.rows}%`,
   744→                }}
   745→              >
   746→                <div className={cn('absolute inset-[5%] board-cell', (r + c) % 2 === 1 && 'board-cell-alt')} />
   747→              </div>
   748→            )
   749→          })
   750→        : null,
   751→    [level.rows, level.cols, size.w],
   752→  )
   753→
   754→  return (
   755→    <div className="relative flex-1 flex flex-col overflow-hidden ww-tap-none">
   756→      {/* backdrop */}
   757→      <div aria-hidden className="absolute inset-0">
   758→        <img src={A(bgKey)} alt="" className="w-full h-full object-cover" draggable={false} />
   759→        <div className="absolute inset-0 bg-gradient-to-b from-[#0e1c14]/45 via-[#0e1c14]/20 to-[#0e1c14]/55" />
   760→      </div>
   761→
   762→      {/* HUD — ribbon row */}
   763→      <header className="relative z-20 flex items-start gap-2 px-3 pt-3">
   764→        <IconButton
   765→          img={A('icon-pause')}
   766→          label="Pause"
   767→          className="mt-1.5"
   768→          onClick={() => {
   769→            discover('entity:rest')
   770→            pausedRef.current = true
   771→            setPaused(true)
   772→          }}
   773→        />
   774→        <div className="flex-1 min-w-0 flex justify-center">
   775→          <RibbonBanner title={stageLabel} subtitle={level.name} size="sm" className="w-full max-w-[300px]" />
   776→        </div>
   777→        <div
   778→          className="hud-pill rounded-2xl px-3 py-1 flex flex-col items-center leading-none mt-0.5 min-w-[64px]"
   779→          aria-label={`${movesLeft} moves left`}
   780→        >
   781→          <span className="text-[9px] uppercase tracking-widest text-[#d9c79a] font-bold">Moves</span>
   782→          <span
   783→            className={cn(
   784→              'font-display text-xl font-extrabold tabular-nums',
   785→              movesLeft <= 3 && 'text-[#ff9d7a] animate-pulse',
   786→            )}
   787→          >
   788→            {movesLeft}
   789→          </span>
   790→        </div>
   791→      </header>
   792→

   794→      <section className="relative z-20 px-3 mt-2 flex items-stretch gap-2" aria-label="Score and objectives">
   795→        <div className="goal-card flex-1 min-w-0 px-2.5 py-1.5 flex flex-wrap items-center gap-x-3 gap-y-1">
   796→          {obj.kind === 'collect'
   797→            ? (obj.collect ?? []).map((goal) => {
   798→                const have = Math.min(collected[goal.type] ?? 0, goal.count)
   799→                const done = have >= goal.count
   800→                return (

   810→                        {done ? '✓ ' : 'Collect '}

   816→                        {have} / {goal.count}

   827→                  <div className="flex items-center gap-1.5 flex-1 min-w-[118px]">
   828→                    <img src={A('star-sparkle')} alt="" className="w-8 h-8 object-contain drop-shadow shrink-0" draggable={false} />
   829→                    <div className="min-w-0 flex-1">
   830→                      <p className="text-[10px] leading-tight text-[#5d3a1a] font-bold truncate">
   831→                        {done ? '✓ ' : ''}Weave {target.toLocaleString()} threads
   832→                      </p>
   833→                      <ProgressBar value={have / target} className="h-2 mt-0.5" />
   834→                      <p className={cn('text-[9px] font-bold tabular-nums', done ? 'text-[#4c8a28]' : 'text-[#7a5c34]')}>
   835→                        {have.toLocaleString()} / {target.toLocaleString()}
   836→                      </p>
   837→                    </div>
   838→                  </div>
   839→                )
   840→              })()}
   841→        </div>
   842→        <div className="hud-pill rounded-2xl px-3 py-1 flex flex-col items-center justify-center min-w-[92px]" aria-live="polite">
   843→          <span className="text-[9px] uppercase tracking-widest text-[#d9c79a] font-bold">Score</span>
   844→          <CountUp value={score} duration={500} className="font-display text-lg font-extrabold" />

   846→            {[0, 1, 2].map((i) => (
   847→              <img
   848→                key={i}
   849→                src={A('star-sparkle')}
   850→                alt=""
   851→                draggable={false}
   852→                className={cn('w-3 h-3 object-contain', i === 0 ? (score > 0 ? '' : 'grayscale opacity-40') : score >= (i === 1 ? level.star2 : level.star3) ? '' : 'grayscale opacity-40')}
   853→              />
   854→            ))}
   855→          </div>
   856→        </div>
   857→      </section>
   858→
   859→      {/* board */}
   860→      <main ref={wrapRef} className="relative z-10 flex-1 min-h-0 flex items-center justify-center px-3 py-2">
   861→        <div
   862→          className={cn('board-frame', shakeId > 0 && 'anim-board-shake')}

   867→            role="application"
   868→            aria-label={`${level.rows} by ${level.cols} weaving board. Drag or tap charms to swap.`}
   869→            className={cn(
   870→              'relative w-full h-full rounded-[0.7rem] overflow-hidden touch-none select-none',
   871→              armedBooster === 'null' && 'cursor-crosshair',
   872→            )}
   873→            style={{
   874→              backgroundImage: 'linear-gradient(180deg, #1c2e1f 0%, #16251a 100%)',
   875→            }}
   876→            onPointerDown={onPointerDown}
   877→            onPointerMove={onPointerMove}
   878→            onPointerUp={onPointerUp}
   879→            onPointerCancel={() => (dragRef.current = null)}
   880→            onContextMenu={(e) => e.preventDefault()}
   881→          >
   882→            {/* cell sockets (memoized) */}
   883→            {sockets}
   884→
   885→            {grid.flatMap((row, r) =>
   886→              row.map((t, c) => {
   887→                if (!t) return null
   888→                return (
   889→                  <TileView
   890→                    key={t.id}
   891→                    tile={t}
   892→                    r={r}
   893→                    c={c}
   894→                    rows={level.rows}
   895→                    cols={level.cols}
   896→                    selected={!!selected && selected.r === r && selected.c === c}
   897→                    hinted={!!hint && hint.some((h) => h.r === r && h.c === c)}
   898→                  />
   899→                )
   900→              }),
   901→            )}
   902→
   903→            {/* floaters */}
   904→            {floaters.map((f) => (
   905→              <div
   906→                key={f.id}
   907→                className="absolute pointer-events-none z-30"
   908→                style={{ left: `${((f.c + 0.5) * 100) / level.cols}%`, top: `${((f.r + 0.5) * 100) / level.rows}%`, width: 0, height: 0 }}

   910→                <span className="anim-rise absolute font-display font-black text-lg text-[#ffe9a8] drop-shadow-[0_2px_2px_rgba(0,0,0,0.8)] whitespace-nowrap -translate-x-1/2">

   912→                </span>
   913→              </div>
   914→            ))}
   915→
   916→            {/* combo banner */}
   917→            {combo && (
   918→              <div key={combo.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   919→                <span className="anim-combo combo-banner font-display italic font-black text-3xl px-6 py-1.5 rounded-full">
   920→                  {combo.text ?? `Combo ×${combo.n}!`}
   921→                </span>
   922→              </div>
   923→            )}
   924→
   925→            {/* objective complete flash */}
   926→            {goalFlash && (
   927→              <div key={goalFlash.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   928→                <img src={A('banner-objective')} alt="Objective complete!" draggable={false} className="anim-combo w-48 drop-shadow-2xl" />
   929→              </div>
   930→            )}
   931→
   932→            {/* toast */}
   933→            {toast && (
   934→              <div className="absolute inset-x-0 bottom-3 z-40 flex justify-center pointer-events-none">
   935→                <span className="hud-pill rounded-full px-4 py-1.5 text-xs font-bold anim-fade-in">{toast}</span>
   936→              </div>
   937→            )}
   938→          </div>
   939→        </div>
   940→      </main>
   941→
   942→      {/* boosters */}
   943→      <footer className="relative z-20 px-4 pb-2 pt-0.5 flex items-center justify-between gap-3">
   944→        <BoosterButton
   945→          kind="lens"
   946→          count={inventory.lens}
   947→          armed={armedBooster === 'lens'}
   948→          disabled={busy || status !== 'playing'}
   949→          onClick={() => armBooster('lens')}
   950→        />
   951→        <div className="flex-1 min-w-0" aria-hidden />
   952→        <BoosterButton
   953→          kind="null"
   954→          count={inventory.null}
   955→          armed={armedBooster === 'null'}
   956→          disabled={busy || status !== 'playing'}
   957→          onClick={() => armBooster('null')}
   958→        />
   959→      </footer>
   960→
   961→      {/* modals */}
   962→      {paused && status === 'playing' && (
   963→        <PauseModal
   964→          level={level}
   965→          isDaily={isDaily}
   966→          onResume={() => {
   967→            pausedRef.current = false
   968→            setPaused(false)
   969→            resetHintTimer()
   970→          }}
   971→          onRestart={() => {
   972→            pausedRef.current = false
   973→            setPaused(false)
   974→            resetLevel()
   975→          }}
   976→          onOptions={() => onOpenInstruments()}
   977→          onQuit={() => {
   978→            pausedRef.current = false
   979→            setPaused(false)
   980→            onExit()
   981→          }}
   982→        />
   983→      )}
   984→      {result && status === 'won' && (
   985→        <FolioSealedModal
   986→          level={level}
   987→          result={result}
   988→          rewards={rewards}
   989→          isDaily={isDaily}
   990→          hasNext={isDaily || hasNextStage}
   991→          onNext={() => (isDaily ? onExit() : onPlayLevel(level.id + 1))}
   992→          onReplay={resetLevel}
   993→          onExit={onExit}
   994→        />
   995→      )}
   996→      {result && status === 'lost' && (
   997→        <FolioLostModal level={level} isDaily={isDaily} score={result.score} onReplay={resetLevel} onExit={onExit} />
   998→      )}
   999→    </div>
  1000→  )
  1001→}

  1048→/* ---------------- tile (memoized — re-renders only when its tile/position/flags change) ---------------- */
  1049→
  1050→const TileView = React.memo(function TileView({
  1051→  tile,
  1052→  r,
  1053→  c,
  1054→  rows,
  1055→  cols,
  1056→  selected,
  1057→  hinted,
  1058→}: {
  1059→  tile: Tile
  1060→  r: number
  1061→  c: number
  1062→  rows: number
  1063→  cols: number
  1064→  selected: boolean
  1065→  hinted: boolean
  1066→}) {
  1067→  const spawned = !!tile.spawned
  1068→  const specialClass =
  1069→    tile.special === 'lineH' || tile.special === 'lineV'
  1070→      ? 'special-lineH'
  1071→      : tile.special === 'bomb'
  1072→        ? 'special-bomb'
  1073→        : ''
  1074→  return (
