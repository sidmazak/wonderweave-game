   331→   331→      // floater at centroid
   332→   332→      if (cellsArr.length > 0) {
   333→   333→        const ar = cellsArr.reduce((s, x) => s + x.r, 0) / cellsArr.length
   334→   334→        const ac = cellsArr.reduce((s, x) => s + x.c, 0) / cellsArr.length
   335→   335→        addFloater(ar, ac, `+${gained}`)
   336→   336→      }
   337→   337→
   338→   338→      // pop
   339→   339→      for (const k of plan.cells) {
   340→   340→        const [r, c] = k.split(',').map(Number)
   341→   341→        const t = g[r]?.[c]
   342→   342→        if (t) t.clearing = true
   343→   343→      }
   344→   344→      setGridBoth(g)
   345→   345→      sfx.pop(cascade)
   346→   346→      vibrate(12)
   347→   347→      await sleep(250)
   348→   348→
   349→   349→      // clear + gravity

   351→   351→      const grav = applyGravity(g, level.types)
   352→   352→      setGridBoth(g)

   356→   356→        let dirty = false

   360→   360→              delete t.spawned
   361→   361→              dirty = true
   362→   362→            }
   363→   363→          }
   364→   364→        }
   365→   365→        if (dirty) setGridBoth(g2)
   366→   366→      } else {
   367→   367→        await sleep(300)
   368→   368→      }
   369→   369→
   370→   370→      // combo banner

   372→   372→

   416→   416→        for (let r = 0; r < level.rows; r++) for (let c = 0; c < level.cols; c++) seed.add(key(r, c))
   417→   417→      } else {
   418→   418→        for (let r = 0; r < level.rows; r++) {
   419→   419→          for (let c = 0; c < level.cols; c++) {
   420→   420→            if (g[r][c]?.type === other.type) seed.add(key(r, c))
   421→   421→          }
   422→   422→        }
   423→   423→      }

   425→   425→      await blast({ cells, promotions: new Map() }, 0)

   427→   427→    },

   447→   447→    }
   448→   448→  }, [cascadeLoop, checkObjective, finishLevel, level.types, setGridBoth, showToast])

   452→   452→    async (a: Cell, b: Cell, opts?: { free?: boolean }): Promise<void> => {
