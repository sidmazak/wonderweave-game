    98→  const [score, setScore] = React.useState(0)
    99→  const scoreRef = React.useRef(0)
   100→  const [movesLeft, setMovesLeft] = React.useState(level.moves)
   101→  const movesRef = React.useRef(level.moves)
   102→  const [collected, setCollected] = React.useState<Record<string, number>>({})
   103→  const collectedRef = React.useRef<Record<string, number>>({})
   104→  const doneGoalsRef = React.useRef<Set<string>>(new Set())
   105→
   106→  const [busy, setBusy] = React.useState(true)
   107→  const busyRef = React.useRef(true)
   108→  const lock = React.useCallback((v: boolean) => {
   109→    busyRef.current = v
   110→    setBusy(v)
   111→  }, [])
   112→
   113→  const [selected, setSelected] = React.useState<Cell | null>(null)
   114→  const selectedRef = React.useRef<Cell | null>(null)
   115→  const setSelectedBoth = React.useCallback((cell: Cell | null) => {
   116→    selectedRef.current = cell
   117→    setSelected(cell)
   118→  }, [])
   119→  const [hint, setHint] = React.useState<[Cell, Cell] | null>(null)
   120→  const hintTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   121→  const dragRef = React.useRef<{ startCell: Cell; x: number; y: number; done: boolean } | null>(null)
   122→
   123→  const [floaters, setFloaters] = React.useState<Floater[]>([])
   124→  const [combo, setCombo] = React.useState<ComboInfo | null>(null)
   125→  const [goalFlash, setGoalFlash] = React.useState<{ id: number } | null>(null)
   126→  const [toast, setToast] = React.useState<string | null>(null)
   127→
   128→  const [status, setStatus] = React.useState<'playing' | 'won' | 'lost'>('playing')
   129→  const statusRef = React.useRef<'playing' | 'won' | 'lost'>('playing')
   130→  const [result, setResult] = React.useState<LevelResult | null>(null)
   131→  const [rewards, setRewards] = React.useState<RewardSummary | null>(null)
   132→  const [paused, setPaused] = React.useState(false)
   133→  const pausedRef = React.useRef(false)
   134→  const [armedBooster, setArmedBooster] = React.useState<BoosterKind | null>(null)
   135→  const armedRef = React.useRef<BoosterKind | null>(null)
   136→  const setArmed = React.useCallback((k: BoosterKind | null) => {
   137→    armedRef.current = k
   138→    setArmedBooster(k)
   139→  }, [])
   140→
   141→  const wrapRef = React.useRef<HTMLDivElement>(null)
   142→  const boardRef = React.useRef<HTMLDivElement>(null)
   143→  const [size, setSize] = React.useState({ w: 0, h: 0 })
   144→
   145→  /* ------- board sizing ------- */
   146→  React.useEffect(() => {
   147→    const el = wrapRef.current
   148→    if (!el) return
   149→    const measure = () => {
   150→      const cs = getComputedStyle(el)
   151→      const availW = el.clientWidth - parseFloat(cs.paddingLeft || '0') - parseFloat(cs.paddingRight || '0')
   152→      const availH = el.clientHeight - parseFloat(cs.paddingTop || '0') - parseFloat(cs.paddingBottom || '0')
   153→      // frame adds 2*3px border + 2*5px padding = 16px; keep a 4px slack
   154→      const target = Math.max(120, Math.min(availW - 20, (availH - 20) * (level.cols / level.rows)))
   155→      setSize({ w: target, h: (target * level.rows) / level.cols })
   156→    }
   157→    measure()
   158→    const ro = new ResizeObserver(measure)
   159→    ro.observe(el)
   160→    return () => ro.disconnect()
   161→  }, [level.rows, level.cols])
   162→
   163→  /* ------- deal animation ------- */
   164→  const clearSpawnedNextFrame = React.useCallback(() => {
   165→    requestAnimationFrame(() =>
   166→      requestAnimationFrame(() => {
   167→        const g = cloneGrid(gridRef.current)
   168→        let dirty = false
   169→        for (const row of g) {
   170→          for (const t of row) {
   171→            if (t && t.spawned) {
   172→              delete t.spawned
   173→              dirty = true
   174→            }
   175→          }
   176→        }
   177→        if (dirty) setGridBoth(g)
   178→      }),
   179→    )
   180→  }, [setGridBoth])
   181→
   182→  React.useEffect(() => {
   183→    initAudio()
   184→    discover('entity:lantern')
   185→    if (level.chapter > 0) discover(`lore:${level.chapter}`)
   186→    clearSpawnedNextFrame()
   187→    const t = setTimeout(() => lock(false), 480 + level.cols * 26)
   188→    return () => clearTimeout(t)
   189→  }, [clearSpawnedNextFrame, lock, level.cols])
   190→
   191→  const resetLevel = React.useCallback(() => {
   192→    const g = createGrid(level.rows, level.cols, level.types)
   193→    for (const row of g) for (const t of row) if (t) t.spawned = true
   194→    setGridBoth(g)
   195→    clearSpawnedNextFrame()
   196→    scoreRef.current = 0
   197→    setScore(0)
   198→    movesRef.current = level.moves
   199→    setMovesLeft(level.moves)
   200→    collectedRef.current = {}
   201→    setCollected({})
   202→    doneGoalsRef.current = new Set()
   203→    setSelectedBoth(null)
   204→    setArmed(null)
   205→    setHint(null)
   206→    setCombo(null)
   207→    setFloaters([])
   208→    setResult(null)
   209→    setRewards(null)
   210→    setStatus('playing')
   211→    statusRef.current = 'playing'
   212→    setPaused(false)
   213→    pausedRef.current = false
   214→    lock(true)
   215→    setTimeout(() => lock(false), 480 + level.cols * 26)
   216→  }, [clearSpawnedNextFrame, level, lock, setArmed, setGridBoth, setSelectedBoth])
   217→
   218→  /* ------- helpers ------- */
   219→  const showToast = React.useCallback((msg: string) => {
   220→    setToast(msg)
   221→    setTimeout(() => setToast((cur) => (cur === msg ? null : cur)), 2200)
   222→  }, [])
   223→
   224→  const addFloater = React.useCallback((r: number, c: number, text: string) => {
   225→    const id = floaterSeq++
   226→    setFloaters((f) => [...f.slice(-6), { id, r, c, text }])
   227→    setTimeout(() => setFloaters((f) => f.filter((x) => x.id !== id)), 1000)
   228→  }, [])
   229→
   230→  const showCombo = React.useCallback((n: number) => {
   231→    const id = comboSeq++
   232→    setCombo({ id, n })
   233→    setTimeout(() => setCombo((cur) => (cur?.id === id ? null : cur)), 1400)
   234→  }, [])
   235→
   236→  const finishLevel = React.useCallback(
   237→    (won: boolean) => {
   238→      const bonus = won ? movesRef.current * 100 : 0
   239→      const finalScore = scoreRef.current + bonus
   240→      const stars = won ? (finalScore >= level.star3 ? 3 : finalScore >= level.star2 ? 2 : 1) : 0
   241→      const res: LevelResult = { levelId: level.id, won, score: finalScore, stars, movesLeft: movesRef.current }
   242→      statusRef.current = won ? 'won' : 'lost'
   243→      setStatus(won ? 'won' : 'lost')
   244→      setResult(res)
   245→      if (won) {
   246→        sfx.win()
   247→        for (let i = 0; i < stars; i++) setTimeout(() => sfx.star(i), 500 + i * 380)
   248→        setRewards(onWin(res))
   249→      } else {
   250→        sfx.lose()
   251→      }
   252→    },
   253→    [level, onWin],
   254→  )
   255→
   256→  const checkObjective = React.useCallback((): boolean => {
   257→    const obj = level.objective
   258→    if (obj.kind === 'score') return scoreRef.current >= (obj.score ?? Infinity)
   259→    return (obj.collect ?? []).every((goal) => (collectedRef.current[goal.type] ?? 0) >= goal.count)
   260→  }, [level])
   261→
   262→  /* ------- blast & cascade ------- */
   263→  const blast = React.useCallback(
   264→    async (plan: ClearPlan, cascade: number): Promise<void> => {
   265→      const g = cloneGrid(gridRef.current)
   266→      const cellsArr = [...plan.cells].map((k) => {
   267→        const [r, c] = k.split(',').map(Number)
   268→        return { r, c }
   269→      })
   270→      if (cellsArr.length === 0 && plan.promotions.size === 0) return
   271→
   272→      const mult = cascade + 1
   273→      const gained = cellsArr.length * 20 * mult + plan.promotions.size * 60
   274→      scoreRef.current += gained
   275→      setScore(scoreRef.current)
   276→
   277→      // collection + codex discovery
   278→      const col = { ...collectedRef.current }
   279→      let lineHit = false
   280→      let bombHit = false
   281→      const seenTypes = new Set<string>()
   282→      for (const { r, c } of cellsArr) {
   283→        const t = g[r]?.[c]
   284→        if (!t) continue
   285→        col[t.type] = (col[t.type] ?? 0) + 1
   286→        seenTypes.add(t.type)
   287→        if (t.special === 'lineH' || t.special === 'lineV') lineHit = true
   288→        if (t.special === 'bomb') bombHit = true
   289→      }
   290→      collectedRef.current = col
   291→      setCollected(col)
   292→      if (seenTypes.size > 0) {
   293→        const lumenIds = [...seenTypes].map((t) => LUMEN_ID_BY_TYPE[t]).filter(Boolean)
   294→        if (lumenIds.length > 0) discover(...lumenIds)
   295→      }
   296→      for (const promo of plan.promotions.values()) {
   297→        if (promo.special === 'prism') discover('lumen:vitriol')
   298→      }
   299→      if (lineHit) sfx.line()
   300→      if (bombHit) sfx.bomb()
   301→
   302→      // floater at centroid
   303→      if (cellsArr.length > 0) {
   304→        const ar = cellsArr.reduce((s, x) => s + x.r, 0) / cellsArr.length
   305→        const ac = cellsArr.reduce((s, x) => s + x.c, 0) / cellsArr.length
   306→        addFloater(ar, ac, `+${gained}`)
   307→      }
   308→
   309→      // pop
   310→      for (const k of plan.cells) {
   311→        const [r, c] = k.split(',').map(Number)
   312→        const t = g[r]?.[c]
   313→        if (t) t.clearing = true
   314→      }
   315→      setGridBoth(g)
   316→      sfx.pop(cascade)
   317→      vibrate(12)
   318→      await sleep(250)
   319→
   320→      // clear + gravity
   321→      applyClear(g, plan)
   322→      const grav = applyGravity(g, level.types)
   323→      setGridBoth(g)
   324→      if (grav.spawnedIds.length > 0) {
   325→        await sleep(430 + level.cols * 26 + 60)
   326→        const g2 = cloneGrid(gridRef.current)
   327→        let dirty = false
   328→        for (const row of g2) {
   329→          for (const t of row) {
   330→            if (t && t.spawned) {
   331→              delete t.spawned
   332→              dirty = true
   333→            }
   334→          }
   335→        }
   336→        if (dirty) setGridBoth(g2)
   337→      } else {
   338→        await sleep(300)
   339→      }
   340→
   341→      // combo banner
   342→      if (mult >= 2) showCombo(mult)
   343→
   344→      // goal flash
   345→      if (level.objective.kind === 'collect') {
   346→        for (const goal of level.objective.collect ?? []) {
   347→          const done = (collectedRef.current[goal.type] ?? 0) >= goal.count
   348→          if (done && !doneGoalsRef.current.has(goal.type)) {
   349→            doneGoalsRef.current.add(goal.type)
   350→            setGoalFlash({ id: Date.now() })
   351→            sfx.specialCreate()
   352→            setTimeout(() => setGoalFlash(null), 1400)
   353→          }
   354→        }
   355→      }
   356→    },
   357→    [addFloater, level, setGridBoth, showCombo, vibrate],
   358→  )
   359→
   360→  const cascadeLoop = React.useCallback(
   361→    async (swapped?: Cell[]): Promise<void> => {
   362→      let cascade = 0
   363→      for (;;) {
   364→        const g = cloneGrid(gridRef.current)
   365→        const shapes = findShapes(g, cascade === 0 ? swapped : undefined)
   366→        if (shapes.length === 0) break
   367→        const plan = planClear(g, shapes, cascade === 0 ? swapped : undefined)
   368→        await blast(plan, cascade)
   369→        cascade++
   370→      }
   371→    },
   372→    [blast],
   373→  )
   374→
   375→  const activatePrism = React.useCallback(
   376→    async (a: Cell, b: Cell): Promise<void> => {
   377→      const g = gridRef.current
   378→      const ta = g[a.r]?.[a.c]
   379→      const tb = g[b.r]?.[b.c]
   380→      if (!ta || !tb) return
   381→      const aIsPrism = ta.special === 'prism'
   382→      const other = aIsPrism ? tb : ta
   383→      sfx.prism()
   384→      discover('lumen:vitriol')
   385→      const seed = new Set<string>()
   386→      if (other.special === 'prism') {
   387→        for (let r = 0; r < level.rows; r++) for (let c = 0; c < level.cols; c++) seed.add(key(r, c))
   388→      } else {
   389→        for (let r = 0; r < level.rows; r++) {
   390→          for (let c = 0; c < level.cols; c++) {
   391→            if (g[r][c]?.type === other.type) seed.add(key(r, c))
   392→          }
   393→        }
   394→      }
   395→      const cells = expandSpecials(g, seed)
   396→      await blast({ cells, promotions: new Map() }, 0)
   397→      await cascadeLoop()
   398→    },
   399→    [blast, cascadeLoop, level.rows, level.cols],
   400→  )
   401→
   402→  const afterMove = React.useCallback(async (): Promise<void> => {
   403→    if (checkObjective()) {
   404→      finishLevel(true)
   405→      return
   406→    }
   407→    if (movesRef.current <= 0) {
   408→      finishLevel(false)
   409→      return
   410→    }
   411→    if (!hasAnyMove(cloneGrid(gridRef.current))) {
   412→      showToast('The loom is stuck — reshuffling threads…')
   413→      sfx.shuffle()
   414→      const ng = shuffleGrid(cloneGrid(gridRef.current), level.types)
   415→      setGridBoth(ng)
   416→      await sleep(450)
   417→      await cascadeLoop()
   418→    }
   419→  }, [cascadeLoop, checkObjective, finishLevel, level.types, setGridBoth, showToast])
   420→
   421→  /* ------- swap flow ------- */
   422→  const attemptSwap = React.useCallback(
   423→    async (a: Cell, b: Cell, opts?: { free?: boolean }): Promise<void> => {
   424→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   425→      if (!a || !b || typeof a.r !== 'number' || typeof b.r !== 'number') return
   426→      if (!adjacent(a, b)) return
   427→      lock(true)
   428→      setSelectedBoth(null)
   429→      setHint(null)
   430→
   431→      const g0 = cloneGrid(gridRef.current)
   432→      const ta = g0[a.r]?.[a.c]
   433→      const tb = g0[b.r]?.[b.c]
   434→      if (!ta || !tb) {
   435→        lock(false)
   436→        return
   437→      }
   438→
   439→      const swapped = cloneGrid(g0)
   440→      swapped[a.r][a.c] = { ...tb }
   441→      swapped[b.r][b.c] = { ...ta }
   442→      setGridBoth(swapped)
   443→      sfx.swap()
   444→      await sleep(190)
   445→
   446→      const g1 = gridRef.current
   447→      const prismInvolved = ta.special === 'prism' || tb.special === 'prism'
   448→      const specialVsSpecial = !prismInvolved && ta.special !== 'none' && tb.special !== 'none'
   449→      // g1 already contains the swapped tiles, so any shape present now was created by the swap
   450→      const valid = prismInvolved || specialVsSpecial || findShapes(g1).length > 0
   451→
   452→      if (!valid) {
   453→        setGridBoth(g0)
   454→        sfx.invalid()
   455→        await sleep(190)
   456→        lock(false)
   457→        return
   458→      }
   459→
   460→      if (!opts?.free) {
   461→        movesRef.current -= 1
   462→        setMovesLeft(movesRef.current)
   463→      }
   464→      vibrate(10)
   465→
   466→      if (prismInvolved) {
   467→        await activatePrism(a, b)
   468→      } else if (specialVsSpecial) {
   469→        const seed = new Set([key(a.r, a.c), key(b.r, b.c)])
   470→        sfx.bomb()
   471→        await blast({ cells: expandSpecials(g1, seed), promotions: new Map() }, 0)
   472→        await cascadeLoop()
   473→      } else {
   474→        await cascadeLoop([a, b])
   475→      }
   476→      await afterMove()
   477→      lock(false)
   478→    },
   479→    [activatePrism, afterMove, blast, cascadeLoop, lock, setGridBoth, setSelectedBoth, vibrate],
   480→  )
   481→
   482→  /* ------- boosters ------- */
   483→  const armBooster = React.useCallback(
   484→    (kind: BoosterKind) => {
   485→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   486→      if (inventory[kind] <= 0) return
   487→      initAudio()
   488→      if (kind === 'lens') {
   489→        // LENS executes the weave instantly (no move spent)
   490→        if (!spendBooster('lens')) return
   491→        setArmed(null)
   492→        setSelectedBoth(null)
   493→        sfx.lens()
   494→        const moves = findAllMoves(cloneGrid(gridRef.current))
   495→        if (moves.length === 0) {
   496→          showToast('No threads available — the loom reshuffles…')
   497→          const ng = shuffleGrid(cloneGrid(gridRef.current), level.types)
   498→          setGridBoth(ng)
   499→          return
   500→        }
   501→        void attemptSwap(moves[0][0], moves[0][1], { free: true })
   502→        return
   503→      }
   504→      sfx.booster()
   505→      setArmed(armedRef.current === 'null' ? null : 'null')
   506→      setSelectedBoth(null)
   507→    },
   508→    [attemptSwap, inventory, level.types, setArmed, setGridBoth, setSelectedBoth, showToast, spendBooster],
   509→  )
   510→
   511→  const nullifyCell = React.useCallback(
   512→    async (cell: Cell): Promise<void> => {
   513→      if (busyRef.current || statusRef.current !== 'playing') return
   514→      if (!spendBooster('null')) {
   515→        setArmed(null)
   516→        return
   517→      }
   518→      lock(true)
   519→      setArmed(null)
   520→      setSelectedBoth(null)
   521→      setHint(null)
   522→      sfx.nullify()
   523→      vibrate(16)
   524→
   525→      const g = cloneGrid(gridRef.current)
   526→      const t = g[cell.r]?.[cell.c]
   527→      if (!t) {
   528→        lock(false)
   529→        return
   530→      }
   531→      t.clearing = true
   532→      setGridBoth(g)
   533→      await sleep(230)
   534→
   535→      applyClear(g, { cells: new Set([key(cell.r, cell.c)]), promotions: new Map() })
   536→      const grav = applyGravity(g, level.types)
   537→      setGridBoth(g)
   538→      if (grav.spawnedIds.length > 0) await sleep(430 + level.cols * 26 + 60)
   539→      else await sleep(280)
   540→      const g2 = cloneGrid(gridRef.current)
   541→      let dirty = false
   542→      for (const row of g2) {
   543→        for (const tl of row) {
   544→          if (tl && tl.spawned) {
   545→            delete tl.spawned
   546→            dirty = true
   547→          }
   548→        }
   549→      }
   550→      if (dirty) setGridBoth(g2)
   551→
   552→      await cascadeLoop()
   553→      await afterMove()
   554→      lock(false)
   555→    },
   556→    [afterMove, cascadeLoop, level.cols, level.types, lock, setArmed, setGridBoth, setSelectedBoth, spendBooster, vibrate],
   557→  )
   558→
   559→  /* ------- pointer input ------- */
   560→  const cellFromPoint = React.useCallback(
   561→    (x: number, y: number): Cell | null => {
   562→      const el = boardRef.current
   563→      if (!el) return null
   564→      const rect = el.getBoundingClientRect()
   565→      const c = Math.floor(((x - rect.left) / rect.width) * level.cols)
   566→      const r = Math.floor(((y - rect.top) / rect.height) * level.rows)
   567→      if (r < 0 || c < 0 || r >= level.rows || c >= level.cols) return null
   568→      return { r, c }
   569→    },
   570→    [level.cols, level.rows],
   571→  )
   572→
   573→  const resetHintTimer = React.useCallback(() => {
   574→    if (hintTimer.current) clearTimeout(hintTimer.current)
   575→    setHint(null)
   576→    hintTimer.current = setTimeout(() => {
   577→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   578→      try {
   579→        const moves = findAllMoves(cloneGrid(gridRef.current))
   580→        if (moves.length > 0) setHint(moves[Math.floor(Math.random() * moves.length)])
   581→      } catch {
   582→        /* never let the hint probe crash the game */
   583→      }
   584→    }, 6000)
   585→  }, [])
   586→
   587→  React.useEffect(() => {
   588→    resetHintTimer()
   589→    return () => {
   590→      if (hintTimer.current) clearTimeout(hintTimer.current)
   591→    }
   592→    // re-arm after activity
   593→  }, [score, movesLeft, busy, resetHintTimer])
   594→
   595→  const onPointerDown = (e: React.PointerEvent) => {
   596→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   597→    const cell = cellFromPoint(e.clientX, e.clientY)
   598→    if (!cell) return
   599→    resetHintTimer()
   600→    if (armedRef.current === 'null') {
   601→      void nullifyCell(cell)
   602→      return
   603→    }
   604→    dragRef.current = { startCell: cell, x: e.clientX, y: e.clientY, done: false }
   605→  }
   606→  const onPointerMove = (e: React.PointerEvent) => {
   607→    const d = dragRef.current
   608→    if (!d || d.done || busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   609→    const dx = e.clientX - d.x
   610→    const dy = e.clientY - d.y
   611→    const el = boardRef.current
   612→    if (!el) return
   613→    const cellPx = el.getBoundingClientRect().width / level.cols
   614→    if (Math.hypot(dx, dy) >= cellPx * 0.35) {
   615→      d.done = true
   616→      const dir =
   617→        Math.abs(dx) > Math.abs(dy)
   618→          ? { dr: 0, dc: dx > 0 ? 1 : -1 }
   619→          : { dr: dy > 0 ? 1 : -1, dc: 0 }
   620→      const target = { r: d.startCell.r + dir.dr, c: d.startCell.c + dir.dc }
   621→      if (target.r >= 0 && target.c >= 0 && target.r < level.rows && target.c < level.cols) {
   622→        setSelectedBoth(null)
   623→        void attemptSwap(d.startCell, target)
   624→      }
   625→    }
   626→  }
   627→
   628→  const onPointerUp = (e: React.PointerEvent) => {
   629→    const d = dragRef.current
   630→    dragRef.current = null
   631→    if (!d || d.done) return
   632→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   633→    const cell = cellFromPoint(e.clientX, e.clientY)
   634→    if (!cell) {
   635→      setSelectedBoth(null)
   636→      return
   637→    }
   638→    if (armedRef.current === 'null') {
   639→      void nullifyCell(cell)
   640→      return
   641→    }
   642→    // pure tap/select logic — side effects happen outside any state updater
   643→    const sel = selectedRef.current
   644→    if (!sel) {
   645→      sfx.select()
   646→      setSelectedBoth(cell)
   647→      return
   648→    }
   649→    if (sel.r === cell.r && sel.c === cell.c) {
   650→      setSelectedBoth(null)
   651→      return
   652→    }
   653→    if (adjacent(sel, cell)) {
   654→      const a = sel
   655→      setSelectedBoth(null)
   656→      void attemptSwap(a, cell)
   657→      return
   658→    }
   659→    sfx.select()
   660→    setSelectedBoth(cell)
   661→  }
   662→
   663→  /* ------- render ------- */
   664→  const obj = level.objective
   665→  const bgKey = level.chapter > 0 ? CHAPTERS[level.chapter - 1].bg : 'bg-altar'
   666→  const stageLabel = isDaily ? 'DAILY PAGE' : `STAGE ${stageTitle(level)}`
   667→
   668→  return (
   669→    <div className="relative flex-1 flex flex-col overflow-hidden ww-tap-none">
   670→      {/* backdrop */}
   671→      <div aria-hidden className="absolute inset-0">
   672→        <img src={A(bgKey)} alt="" className="w-full h-full object-cover" draggable={false} />
   673→        <div className="absolute inset-0 bg-gradient-to-b from-[#0e1c14]/45 via-[#0e1c14]/20 to-[#0e1c14]/55" />
   674→      </div>
   675→
   676→      {/* HUD — ribbon row */}
   677→      <header className="relative z-20 flex items-start gap-2 px-3 pt-3">
   678→        <IconButton
   679→          img={A('icon-pause')}
   680→          label="Pause"
   681→          className="mt-1.5"
   682→          onClick={() => {
   683→            discover('entity:rest')
   684→            pausedRef.current = true
   685→            setPaused(true)
   686→          }}
   687→        />
   688→        <div className="flex-1 min-w-0 flex justify-center">
   689→          <RibbonBanner title={stageLabel} subtitle={level.name} size="sm" className="w-full max-w-[300px]" />
   690→        </div>
   691→        <div
   692→          className="hud-pill rounded-2xl px-3 py-1 flex flex-col items-center leading-none mt-0.5 min-w-[64px]"
   693→          aria-label={`${movesLeft} moves left`}
   694→        >
   695→          <span className="text-[9px] uppercase tracking-widest text-[#d9c79a] font-bold">Moves</span>
   696→          <span
   697→            className={cn(
   698→              'font-display text-xl font-extrabold tabular-nums',
   699→              movesLeft <= 3 && 'text-[#ff9d7a] animate-pulse',
   700→            )}
   701→          >
   702→            {movesLeft}
   703→          </span>
   704→        </div>
   705→      </header>
   706→
   707→      {/* goals + score */}
   708→      <section className="relative z-20 px-3 mt-2 flex items-stretch gap-2" aria-label="Score and objectives">
   709→        <div className="goal-card flex-1 min-w-0 px-2.5 py-1.5 flex flex-wrap items-center gap-x-3 gap-y-1">
   710→          {obj.kind === 'collect'
   711→            ? (obj.collect ?? []).map((goal) => {
   712→                const have = Math.min(collected[goal.type] ?? 0, goal.count)
   713→                const done = have >= goal.count
   714→                return (
   715→                  <div key={goal.type} className="flex items-center gap-1.5 min-w-[118px] flex-1">
   716→                    <img
   717→                      src={TILE_IMG[goal.type]}
   718→                      alt={TILE_META[goal.type].name}
   719→                      className="w-8 h-8 object-contain drop-shadow shrink-0"
   720→                      draggable={false}
   721→                    />
   722→                    <div className="min-w-0 flex-1">
   723→                      <p className="text-[10px] leading-tight text-[#5d3a1a] font-bold truncate">
   724→                        {done ? '✓ ' : 'Collect '}
   725→                        {goal.count} {TILE_META[goal.type].name}
   726→                        {goal.count > 1 ? '' : ''}
   727→                      </p>
   728→                      <ProgressBar value={have / goal.count} className="h-2 mt-0.5" />
   729→                      <p className={cn('text-[9px] font-bold tabular-nums', done ? 'text-[#4c8a28]' : 'text-[#7a5c34]')}>
   730→                        {have} / {goal.count}
   731→                      </p>
   732→                    </div>
   733→                  </div>
   734→                )
   735→              })
   736→            : (() => {
   737→                const target = obj.score ?? level.star3
   738→                const have = Math.min(score, target)
   739→                const done = score >= target
   740→                return (
   741→                  <div className="flex items-center gap-1.5 flex-1 min-w-[118px]">
   742→                    <img src={A('star-sparkle')} alt="" className="w-8 h-8 object-contain drop-shadow shrink-0" draggable={false} />
   743→                    <div className="min-w-0 flex-1">
   744→                      <p className="text-[10px] leading-tight text-[#5d3a1a] font-bold truncate">
   745→                        {done ? '✓ ' : ''}Weave {target.toLocaleString()} threads
   746→                      </p>
   747→                      <ProgressBar value={have / target} className="h-2 mt-0.5" />
   748→                      <p className={cn('text-[9px] font-bold tabular-nums', done ? 'text-[#4c8a28]' : 'text-[#7a5c34]')}>
   749→                        {have.toLocaleString()} / {target.toLocaleString()}
   750→                      </p>
   751→                    </div>
   752→                  </div>
   753→                )
   754→              })()}
   755→        </div>
   756→        <div className="hud-pill rounded-2xl px-3 py-1 flex flex-col items-center justify-center min-w-[92px]" aria-live="polite">
   757→          <span className="text-[9px] uppercase tracking-widest text-[#d9c79a] font-bold">Score</span>
   758→          <CountUp value={score} duration={500} className="font-display text-lg font-extrabold" />
   759→          <div className="flex gap-0.5 mt-0.5" aria-hidden>
   760→            {[0, 1, 2].map((i) => (
   761→              <img
   762→                key={i}
   763→                src={A('star-sparkle')}
   764→                alt=""
   765→                draggable={false}
   766→                className={cn('w-3 h-3 object-contain', i === 0 ? (score > 0 ? '' : 'grayscale opacity-40') : score >= (i === 1 ? level.star2 : level.star3) ? '' : 'grayscale opacity-40')}
   767→              />
   768→            ))}
   769→          </div>
   770→        </div>
   771→      </section>
   772→
   773→      {/* board */}
   774→      <main ref={wrapRef} className="relative z-10 flex-1 min-h-0 flex items-center justify-center px-3 py-2">
   775→        <div
   776→          className="board-frame"
   777→          style={{ width: size.w + 12, height: size.h + 12, padding: 5, opacity: size.w > 0 ? 1 : 0 }}
   778→        >
   779→          <div
   780→            ref={boardRef}
   781→            role="application"
   782→            aria-label={`${level.rows} by ${level.cols} weaving board. Drag or tap charms to swap.`}
   783→            className={cn(
   784→              'relative w-full h-full rounded-[0.7rem] overflow-hidden touch-none select-none',
   785→              armedBooster === 'null' && 'cursor-crosshair',
   786→            )}
   787→            style={{
   788→              backgroundImage: 'linear-gradient(180deg, #1c2e1f 0%, #16251a 100%)',
   789→            }}
   790→            onPointerDown={onPointerDown}
   791→            onPointerMove={onPointerMove}
   792→            onPointerUp={onPointerUp}
   793→            onPointerCancel={() => (dragRef.current = null)}
   794→            onContextMenu={(e) => e.preventDefault()}
   795→          >
   796→            {/* cell sockets */}
   797→            {size.w > 0 &&
   798→              Array.from({ length: level.rows * level.cols }).map((_, i) => {
   799→                const r = Math.floor(i / level.cols)
   800→                const c = i % level.cols
   801→                return (
   802→                  <div
   803→                    key={i}
   804→                    aria-hidden
   805→                    className="absolute"
   806→                    style={{
   807→                      left: `${(c * 100) / level.cols}%`,
   808→                      top: `${(r * 100) / level.rows}%`,
   809→                      width: `${100 / level.cols}%`,
   810→                      height: `${100 / level.rows}%`,
   811→                    }}
   812→                  >
   813→                    <div className={cn('absolute inset-[5%] board-cell', (r + c) % 2 === 1 && 'board-cell-alt')} />
   814→                  </div>
   815→                )
   816→              })}
   817→
   818→            {grid.flatMap((row, r) =>
   819→              row.map((t, c) => {
   820→                if (!t) return null
   821→                return (
   822→                  <TileView
   823→                    key={t.id}
   824→                    tile={t}
   825→                    r={r}
   826→                    c={c}
   827→                    rows={level.rows}
   828→                    cols={level.cols}
   829→                    selected={!!selected && selected.r === r && selected.c === c}
   830→                    hinted={!!hint && hint.some((h) => h.r === r && h.c === c)}
   831→                  />
   832→                )
   833→              }),
   834→            )}
   835→
   836→            {/* floaters */}
   837→            {floaters.map((f) => (
   838→              <div
   839→                key={f.id}
   840→                className="absolute pointer-events-none z-30"
   841→                style={{ left: `${((f.c + 0.5) * 100) / level.cols}%`, top: `${((f.r + 0.5) * 100) / level.rows}%`, width: 0, height: 0 }}
   842→              >
   843→                <span className="anim-rise absolute font-display font-black text-lg text-[#ffe9a8] drop-shadow-[0_2px_2px_rgba(0,0,0,0.8)] whitespace-nowrap -translate-x-1/2">
   844→                  {f.text}
   845→                </span>
   846→              </div>
   847→            ))}
   848→
   849→            {/* combo banner */}
   850→            {combo && (
   851→              <div key={combo.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   852→                <span className="anim-combo font-display italic font-black text-3xl text-[#ffd76e] px-6 py-1.5 rounded-full border-2 border-[#f4c05a] bg-gradient-to-b from-[#2c4a6e]/95 to-[#16263e]/95 drop-shadow-[0_4px_10px_rgba(0,0,0,0.55)]">
   853→                  Combo ×{combo.n}!
   854→                </span>
   855→              </div>
   856→            )}
   857→
   858→            {/* objective complete flash */}
   859→            {goalFlash && (
   860→              <div key={goalFlash.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   861→                <img src={A('banner-objective')} alt="Objective complete!" draggable={false} className="anim-combo w-48 drop-shadow-2xl" />
   862→              </div>
   863→            )}
   864→
   865→            {/* toast */}
   866→            {toast && (
   867→              <div className="absolute inset-x-0 bottom-3 z-40 flex justify-center pointer-events-none">
   868→                <span className="hud-pill rounded-full px-4 py-1.5 text-xs font-bold anim-fade-in">{toast}</span>
   869→              </div>
   870→            )}
   871→          </div>
   872→        </div>
   873→      </main>
   874→
   875→      {/* boosters + hint */}
   876→      <footer className="relative z-20 px-4 pb-2 pt-0.5 flex items-center justify-between gap-3">
   877→        <BoosterButton
   878→          kind="lens"
   879→          count={inventory.lens}
   880→          armed={armedBooster === 'lens'}
   881→          disabled={busy || status !== 'playing'}
   882→          onClick={() => armBooster('lens')}
   883→        />
   884→        <p className="flex-1 min-w-0 text-center text-[10px] leading-tight text-[#f4e9c8]/85 ww-text-outline font-medium">
   885→          {armedBooster === 'null' ? 'Tap any charm to unweave it…' : level.hint}
   886→        </p>
   887→        <BoosterButton
   888→          kind="null"
   889→          count={inventory.null}
   890→          armed={armedBooster === 'null'}
   891→          disabled={busy || status !== 'playing'}
   892→          onClick={() => armBooster('null')}
   893→        />
   894→      </footer>
   895→
   896→      {/* modals */}
   897→      {paused && status === 'playing' && (
   898→        <PauseModal
   899→          level={level}
   900→          isDaily={isDaily}
   901→          onResume={() => {
   902→            pausedRef.current = false
   903→            setPaused(false)
   904→            resetHintTimer()
   905→          }}
   906→          onRestart={() => {
   907→            pausedRef.current = false
   908→            setPaused(false)
   909→            resetLevel()
   910→          }}
   911→          onOptions={() => onOpenInstruments()}
   912→          onQuit={() => {
   913→            pausedRef.current = false
   914→            setPaused(false)
   915→            onExit()
   916→          }}
   917→        />
   918→      )}
   919→      {result && status === 'won' && (
   920→        <FolioSealedModal
   921→          level={level}
   922→          result={result}
   923→          rewards={rewards}
   924→          isDaily={isDaily}
   925→          hasNext={isDaily || hasNextStage}
   926→          onNext={() => (isDaily ? onExit() : onPlayLevel(level.id + 1))}
   927→          onReplay={resetLevel}
   928→          onExit={onExit}
   929→        />
   930→      )}
   931→      {result && status === 'lost' && (
   932→        <FolioLostModal level={level} isDaily={isDaily} score={result.score} onReplay={resetLevel} onExit={onExit} />
   933→      )}
   934→    </div>
   935→  )
   936→}
