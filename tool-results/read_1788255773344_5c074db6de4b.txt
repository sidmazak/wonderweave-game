    35→  return Math.abs(a.r - b.r) + Math.abs(a.c - b.c) === 1

    52→export function PlayScreen({
    53→  levelId,
    54→  playerName,
    55→  sound,
    56→  vibration,
    57→  onSound,
    58→  onVibration,
    59→  onExitToAtlas,
    60→  onPlayLevel,
    61→  onSaveResult,
    62→}: {
    63→  levelId: number
    64→  playerName: string
    65→  sound: boolean
    66→  vibration: boolean
    67→  onSound: (v: boolean) => void
    68→  onVibration: (v: boolean) => void
    69→  onExitToAtlas: () => void
    70→  onPlayLevel: (id: number) => void
    71→  onSaveResult: (result: LevelResult) => void
    72→}) {
    73→  const level = React.useMemo(() => getLevel(levelId), [levelId])
    74→
    75→  const [grid, setGrid] = React.useState<Grid>(() => {
    76→    const g = createGrid(level.rows, level.cols, level.types)
    77→    for (const row of g) for (const t of row) if (t) t.spawned = true
    78→    return g
    79→  })
    80→  const gridRef = React.useRef(grid)
    81→  const setGridBoth = React.useCallback((g: Grid) => {
    82→    gridRef.current = g
    83→    setGrid(g)
    84→  }, [])
    85→
    86→  const [score, setScore] = React.useState(0)
    87→  const scoreRef = React.useRef(0)
    88→  const [movesLeft, setMovesLeft] = React.useState(level.moves)
    89→  const movesRef = React.useRef(level.moves)
    90→  const [collected, setCollected] = React.useState<Record<string, number>>({})
    91→  const collectedRef = React.useRef<Record<string, number>>({})
    92→  const doneGoalsRef = React.useRef<Set<string>>(new Set())
    93→
    94→  const [busy, setBusy] = React.useState(true)
    95→  const busyRef = React.useRef(true)
    96→  const lock = React.useCallback((v: boolean) => {
    97→    busyRef.current = v
    98→    setBusy(v)
    99→  }, [])
   100→
   101→  const [selected, setSelected] = React.useState<Cell | null>(null)
   102→  const selectedRef = React.useRef<Cell | null>(null)
   103→  const setSelectedBoth = React.useCallback((cell: Cell | null) => {
   104→    selectedRef.current = cell
   105→    setSelected(cell)
   106→  }, [])
   107→  const [hint, setHint] = React.useState<[Cell, Cell] | null>(null)
   108→  const hintTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null)
   109→  const dragRef = React.useRef<{ startCell: Cell; x: number; y: number; done: boolean } | null>(null)
   110→
   111→  const [floaters, setFloaters] = React.useState<Floater[]>([])
   112→  const [combo, setCombo] = React.useState<ComboInfo | null>(null)
   113→  const [goalFlash, setGoalFlash] = React.useState<{ id: number } | null>(null)
   114→  const [toast, setToast] = React.useState<string | null>(null)
   115→
   116→  const [status, setStatus] = React.useState<'playing' | 'won' | 'lost'>('playing')
   117→  const statusRef = React.useRef<'playing' | 'won' | 'lost'>('playing')
   118→  const [result, setResult] = React.useState<LevelResult | null>(null)
   119→  const [paused, setPaused] = React.useState(false)
   120→  const pausedRef = React.useRef(false)
   121→  const [showOptions, setShowOptions] = React.useState(false)
   122→  const runIdRef = React.useRef(0)
   123→
   124→  const wrapRef = React.useRef<HTMLDivElement>(null)
   125→  const boardRef = React.useRef<HTMLDivElement>(null)
   126→  const [size, setSize] = React.useState({ w: 0, h: 0 })
   127→
   128→  React.useEffect(() => {
   129→    setMuted(!sound)
   130→  }, [sound])
   131→
   132→  const vibrate = React.useCallback(
   133→    (ms: number) => {
   134→      if (vibration && typeof navigator !== 'undefined' && typeof navigator.vibrate === 'function') {
   135→        navigator.vibrate(ms)
   136→      }
   137→    },
   138→    [vibration],
   139→  )
   140→
   141→  /* ------- board sizing ------- */
   142→  React.useEffect(() => {
   143→    const el = wrapRef.current
   144→    if (!el) return
   145→    const measure = () => {
   146→      const cs = getComputedStyle(el)
   147→      const availW = el.clientWidth - parseFloat(cs.paddingLeft || '0') - parseFloat(cs.paddingRight || '0')
   148→      const availH = el.clientHeight - parseFloat(cs.paddingTop || '0') - parseFloat(cs.paddingBottom || '0')
   149→      // frame adds 2*3px border + 2*5px padding = 16px; keep a 4px slack
   150→      const target = Math.max(120, Math.min(availW - 20, (availH - 20) * (level.cols / level.rows)))
   151→      setSize({ w: target, h: (target * level.rows) / level.cols })
   152→    }
   153→    measure()
   154→    const ro = new ResizeObserver(measure)
   155→    ro.observe(el)
   156→    return () => ro.disconnect()
   157→  }, [level.rows, level.cols])
   158→
   159→  /* ------- deal animation ------- */
   160→  const clearSpawnedNextFrame = React.useCallback(() => {
   161→    requestAnimationFrame(() =>
   162→      requestAnimationFrame(() => {
   163→        const g = cloneGrid(gridRef.current)
   164→        let dirty = false
   165→        for (const row of g) {
   166→          for (const t of row) {
   167→            if (t && t.spawned) {
   168→              delete t.spawned
   169→              dirty = true
   170→            }
   171→          }
   172→        }
   173→        if (dirty) setGridBoth(g)
   174→      }),
   175→    )
   176→  }, [setGridBoth])
   177→
   178→  React.useEffect(() => {
   179→    initAudio()
   180→    clearSpawnedNextFrame()
   181→    const t = setTimeout(() => lock(false), 480 + level.cols * 26)
   182→    return () => clearTimeout(t)
   183→  }, [clearSpawnedNextFrame, lock, level.cols])
   184→
   185→  const resetLevel = React.useCallback(() => {
   186→    runIdRef.current++
   187→    const g = createGrid(level.rows, level.cols, level.types)
   188→    for (const row of g) for (const t of row) if (t) t.spawned = true
   189→    setGridBoth(g)
   190→    clearSpawnedNextFrame()
   191→    scoreRef.current = 0
   192→    setScore(0)
   193→    movesRef.current = level.moves
   194→    setMovesLeft(level.moves)
   195→    collectedRef.current = {}
   196→    setCollected({})
   197→    doneGoalsRef.current = new Set()
   198→    setSelectedBoth(null)
   199→    setHint(null)
   200→    setCombo(null)
   201→    setFloaters([])
   202→    setResult(null)
   203→    setStatus('playing')
   204→    statusRef.current = 'playing'
   205→    setPaused(false)
   206→    pausedRef.current = false
   207→    lock(true)
   208→    setTimeout(() => lock(false), 480 + level.cols * 26)
   209→  }, [clearSpawnedNextFrame, level, lock, setGridBoth, setSelectedBoth])
   210→
   211→  /* ------- helpers ------- */
   212→  const showToast = React.useCallback((msg: string) => {
   213→    setToast(msg)
   214→    setTimeout(() => setToast((cur) => (cur === msg ? null : cur)), 2200)
   215→  }, [])
   216→
   217→  const addFloater = React.useCallback((r: number, c: number, text: string) => {
   218→    const id = floaterSeq++
   219→    setFloaters((f) => [...f.slice(-6), { id, r, c, text }])
   220→    setTimeout(() => setFloaters((f) => f.filter((x) => x.id !== id)), 1000)
   221→  }, [])
   222→
   223→  const showCombo = React.useCallback((n: number) => {
   224→    const id = comboSeq++
   225→    setCombo({ id, n })
   226→    setTimeout(() => setCombo((cur) => (cur?.id === id ? null : cur)), 1400)
   227→  }, [])
   228→
   229→  const finishLevel = React.useCallback(
   230→    (won: boolean) => {
   231→      const bonus = won ? movesRef.current * 100 : 0
   232→      const finalScore = scoreRef.current + bonus
   233→      const stars = won ? (finalScore >= level.star3 ? 3 : finalScore >= level.star2 ? 2 : 1) : 0
   234→      const res: LevelResult = { levelId: level.id, won, score: finalScore, stars, movesLeft: movesRef.current }
   235→      statusRef.current = won ? 'won' : 'lost'
   236→      setStatus(won ? 'won' : 'lost')
   237→      setResult(res)
   238→      if (won) {
   239→        sfx.win()
   240→        for (let i = 0; i < stars; i++) setTimeout(() => sfx.star(i), 500 + i * 380)
   241→        onSaveResult(res)
   242→      } else {
   243→        sfx.lose()
   244→      }
   245→    },
   246→    [level, onSaveResult],
   247→  )
   248→
   249→  const checkObjective = React.useCallback((): boolean => {
   250→    const obj = level.objective
   251→    if (obj.kind === 'score') return scoreRef.current >= (obj.score ?? Infinity)
   252→    return (obj.collect ?? []).every((goal) => (collectedRef.current[goal.type] ?? 0) >= goal.count)
   253→  }, [level])
   254→
   255→  /* ------- blast & cascade ------- */
   256→  const blast = React.useCallback(
   257→    async (plan: ClearPlan, cascade: number): Promise<void> => {
   258→      const g = cloneGrid(gridRef.current)
   259→      const cellsArr = [...plan.cells].map((k) => {
   260→        const [r, c] = k.split(',').map(Number)
   261→        return { r, c }
   262→      })
   263→      if (cellsArr.length === 0 && plan.promotions.size === 0) return
   264→
   265→      const mult = cascade + 1
   266→      const gained = cellsArr.length * 20 * mult + plan.promotions.size * 60
   267→      scoreRef.current += gained
   268→      setScore(scoreRef.current)
   269→
   270→      // collection + special sfx
   271→      const col = { ...collectedRef.current }
   272→      let lineHit = false
   273→      let bombHit = false
   274→      for (const { r, c } of cellsArr) {
   275→        const t = g[r]?.[c]
   276→        if (!t) continue
   277→        col[t.type] = (col[t.type] ?? 0) + 1
   278→        if (t.special === 'lineH' || t.special === 'lineV') lineHit = true
   279→        if (t.special === 'bomb') bombHit = true
   280→      }
   281→      collectedRef.current = col
   282→      setCollected(col)
   283→      if (lineHit) sfx.line()
   284→      if (bombHit) sfx.bomb()
   285→
   286→      // floater at centroid
   287→      if (cellsArr.length > 0) {
   288→        const ar = cellsArr.reduce((s, x) => s + x.r, 0) / cellsArr.length
   289→        const ac = cellsArr.reduce((s, x) => s + x.c, 0) / cellsArr.length
   290→        addFloater(ar, ac, `+${gained}`)
   291→      }
   292→
   293→      // pop
   294→      for (const k of plan.cells) {
   295→        const [r, c] = k.split(',').map(Number)
   296→        const t = g[r]?.[c]
   297→        if (t) t.clearing = true
   298→      }
   299→      setGridBoth(g)
   300→      sfx.pop(cascade)
   301→      vibrate(12)
   302→      await sleep(250)
   303→
   304→      // clear + gravity
   305→      applyClear(g, plan)
   306→      const grav = applyGravity(g, level.types)
   307→      setGridBoth(g)
   308→      if (grav.spawnedIds.length > 0) {
   309→        await sleep(430 + level.cols * 26 + 60)
   310→        const g2 = cloneGrid(gridRef.current)
   311→        let dirty = false
   312→        for (const row of g2) {
   313→          for (const t of row) {
   314→            if (t && t.spawned) {
   315→              delete t.spawned
   316→              dirty = true
   317→            }
   318→          }
   319→        }
   320→        if (dirty) setGridBoth(g2)
   321→      } else {
   322→        await sleep(300)
   323→      }
   324→
   325→      // combo banner
   326→      if (mult >= 2) showCombo(mult)
   327→
   328→      // goal flash
   329→      if (level.objective.kind === 'collect') {
   330→        for (const goal of level.objective.collect ?? []) {
   331→          const done = (collectedRef.current[goal.type] ?? 0) >= goal.count
   332→          if (done && !doneGoalsRef.current.has(goal.type)) {
   333→            doneGoalsRef.current.add(goal.type)
   334→            setGoalFlash({ id: Date.now() })
   335→            sfx.specialCreate()
   336→            setTimeout(() => setGoalFlash(null), 1400)
   337→          }
   338→        }
   339→      }
   340→    },
   341→    [addFloater, level, setGridBoth, showCombo, vibrate],
   342→  )
   343→
   344→  const cascadeLoop = React.useCallback(
   345→    async (swapped?: Cell[]): Promise<void> => {
   346→      let cascade = 0
   347→      for (;;) {
   348→        const g = cloneGrid(gridRef.current)
   349→        const shapes = findShapes(g, cascade === 0 ? swapped : undefined)
   350→        if (shapes.length === 0) break
   351→        const plan = planClear(g, shapes, cascade === 0 ? swapped : undefined)
   352→        await blast(plan, cascade)
   353→        cascade++
   354→      }
   355→    },
   356→    [blast],
   357→  )
   358→
   359→  const activatePrism = React.useCallback(
   360→    async (a: Cell, b: Cell): Promise<void> => {
   361→      const g = gridRef.current
   362→      const ta = g[a.r][a.c]
   363→      const tb = g[b.r][b.c]
   364→      if (!ta || !tb) return
   365→      const aIsPrism = ta.special === 'prism'
   366→      const other = aIsPrism ? tb : ta
   367→      sfx.prism()
   368→      const seed = new Set<string>()
   369→      if (other.special === 'prism') {
   370→        // prism + prism: weave the whole loom
   371→        for (let r = 0; r < level.rows; r++) for (let c = 0; c < level.cols; c++) seed.add(key(r, c))
   372→      } else {
   373→        for (let r = 0; r < level.rows; r++) {
   374→          for (let c = 0; c < level.cols; c++) {
   375→            if (g[r][c]?.type === other.type) seed.add(key(r, c))
   376→          }
   377→        }
   378→      }
   379→      const cells = expandSpecials(g, seed)
   380→      await blast({ cells, promotions: new Map() }, 0)
   381→      await cascadeLoop()
   382→    },
   383→    [blast, cascadeLoop, level.rows, level.cols],
   384→  )
   385→
   386→  const afterMove = React.useCallback(async (): Promise<void> => {
   387→    if (checkObjective()) {
   388→      finishLevel(true)
   389→      return
   390→    }
   391→    if (movesRef.current <= 0) {
   392→      finishLevel(false)
   393→      return
   394→    }
   395→    if (!hasAnyMove(cloneGrid(gridRef.current))) {
   396→      showToast('The loom is stuck — reshuffling threads…')
   397→      sfx.shuffle()
   398→      const ng = shuffleGrid(cloneGrid(gridRef.current), level.types)
   399→      setGridBoth(ng)
   400→      await sleep(450)
   401→      // a shuffle may create matches — resolve them freely
   402→      await cascadeLoop()
   403→    }
   404→  }, [cascadeLoop, checkObjective, finishLevel, level.types, setGridBoth, showToast])
   405→
   406→  /* ------- swap flow ------- */
   407→  const attemptSwap = React.useCallback(
   408→    async (a: Cell, b: Cell): Promise<void> => {
   409→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   410→      if (!a || !b || typeof a.r !== 'number' || typeof b.r !== 'number') return
   411→      if (!adjacent(a, b)) return
   412→      lock(true)
   413→      setSelectedBoth(null)
   414→      setHint(null)
   415→
   416→      const g0 = cloneGrid(gridRef.current)
   417→      const ta = g0[a.r]?.[a.c]
   418→      const tb = g0[b.r]?.[b.c]
   419→      if (!ta || !tb) {
   420→        lock(false)
   421→        return
   422→      }
   423→
   424→      const swapped = cloneGrid(g0)
   425→      swapped[a.r][a.c] = { ...tb }
   426→      swapped[b.r][b.c] = { ...ta }
   427→      setGridBoth(swapped)
   428→      sfx.swap()
   429→      await sleep(190)
   430→
   431→      const g1 = gridRef.current
   432→      const prismInvolved = ta.special === 'prism' || tb.special === 'prism'
   433→      const specialVsSpecial = !prismInvolved && ta.special !== 'none' && tb.special !== 'none'
   434→      // g1 already contains the swapped tiles, so any shape present now was created by the swap
   435→      const valid = prismInvolved || specialVsSpecial || findShapes(g1).length > 0
   436→
   437→      if (!valid) {
   438→        setGridBoth(g0)
   439→        sfx.invalid()
   440→        await sleep(190)
   441→        lock(false)
   442→        return
   443→      }
   444→
   445→      movesRef.current -= 1
   446→      setMovesLeft(movesRef.current)
   447→      vibrate(10)
   448→
   449→      if (prismInvolved) {
   450→        await activatePrism(a, b)
   451→      } else if (specialVsSpecial) {
   452→        const seed = new Set([key(a.r, a.c), key(b.r, b.c)])
   453→        sfx.bomb()
   454→        await blast({ cells: expandSpecials(g1, seed), promotions: new Map() }, 0)
   455→        await cascadeLoop()
   456→      } else {
   457→        await cascadeLoop([a, b])
   458→      }
   459→      await afterMove()
   460→      lock(false)
   461→    },
   462→    [activatePrism, afterMove, blast, cascadeLoop, lock, setGridBoth, setSelectedBoth, vibrate],
   463→  )
   464→
   465→  /* ------- pointer input ------- */
   466→  const cellFromPoint = React.useCallback(
   467→    (x: number, y: number): Cell | null => {
   468→      const el = boardRef.current
   469→      if (!el) return null
   470→      const rect = el.getBoundingClientRect()
   471→      const c = Math.floor(((x - rect.left) / rect.width) * level.cols)
   472→      const r = Math.floor(((y - rect.top) / rect.height) * level.rows)
   473→      if (r < 0 || c < 0 || r >= level.rows || c >= level.cols) return null
   474→      return { r, c }
   475→    },
   476→    [level.cols, level.rows],
   477→  )
   478→
   479→  const resetHintTimer = React.useCallback(() => {
   480→    if (hintTimer.current) clearTimeout(hintTimer.current)
   481→    setHint(null)
   482→    hintTimer.current = setTimeout(() => {
   483→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   484→      try {
   485→        const moves = findAllMoves(cloneGrid(gridRef.current))
   486→        if (moves.length > 0) setHint(moves[Math.floor(Math.random() * moves.length)])
   487→      } catch {
   488→        /* never let the hint probe crash the game */
   489→      }
   490→    }, 6000)
   491→  }, [])
   492→
   493→  React.useEffect(() => {
   494→    resetHintTimer()
   495→    return () => {
   496→      if (hintTimer.current) clearTimeout(hintTimer.current)
   497→    }
   498→    // re-arm after activity
   499→  }, [score, movesLeft, busy, resetHintTimer])
   500→
   501→  const onPointerDown = (e: React.PointerEvent) => {
   502→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   503→    const cell = cellFromPoint(e.clientX, e.clientY)
   504→    if (!cell) return
   505→    resetHintTimer()
   506→    dragRef.current = { startCell: cell, x: e.clientX, y: e.clientY, done: false }
   507→  }
   508→  const onPointerMove = (e: React.PointerEvent) => {
   509→    const d = dragRef.current
   510→    if (!d || d.done || busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   511→    const dx = e.clientX - d.x
   512→    const dy = e.clientY - d.y
   513→    const el = boardRef.current
   514→    if (!el) return
   515→    const cellPx = el.getBoundingClientRect().width / level.cols
   516→    if (Math.hypot(dx, dy) >= cellPx * 0.35) {
   517→      d.done = true
   518→      const dir =
   519→        Math.abs(dx) > Math.abs(dy)
   520→          ? { dr: 0, dc: dx > 0 ? 1 : -1 }
   521→          : { dr: dy > 0 ? 1 : -1, dc: 0 }
   522→      const target = { r: d.startCell.r + dir.dr, c: d.startCell.c + dir.dc }
   523→      if (target.r >= 0 && target.c >= 0 && target.r < level.rows && target.c < level.cols) {
   524→        setSelectedBoth(null)
   525→        void attemptSwap(d.startCell, target)
   526→      }
   527→    }
   528→  }
   529→
   530→  const onPointerUp = (e: React.PointerEvent) => {
   531→    const d = dragRef.current
   532→    dragRef.current = null
   533→    if (!d || d.done) return
   534→    if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   535→    const cell = cellFromPoint(e.clientX, e.clientY)
   536→    if (!cell) {
   537→      setSelectedBoth(null)
   538→      return
   539→    }
   540→    // pure tap/select logic — side effects happen outside any state updater
   541→    const sel = selectedRef.current
   542→    if (!sel) {
   543→      sfx.select()
   544→      setSelectedBoth(cell)
   545→      return
   546→    }
   547→    if (sel.r === cell.r && sel.c === cell.c) {
   548→      setSelectedBoth(null)
   549→      return
   550→    }
   551→    if (adjacent(sel, cell)) {
   552→      const a = sel
   553→      setSelectedBoth(null)
   554→      void attemptSwap(a, cell)
   555→      return
   556→    }
   557→    sfx.select()
   558→    setSelectedBoth(cell)
   559→  }
   560→
   561→  /* ------- render ------- */
   562→  const obj = level.objective
   563→  const star3 = level.star3
   564→  const fill = Math.min(1, score / star3)
   565→  const won = status === 'won'
   566→
   567→  return (
   568→    <div className="relative flex-1 flex flex-col overflow-hidden">
   569→      {/* backdrop */}
   570→      <div aria-hidden className="absolute inset-0">
   571→        <img src={A('bg-forest')} alt="" className="w-full h-full object-cover" draggable={false} />
   572→        <div className="absolute inset-0 bg-gradient-to-b from-[#0e1c14]/35 via-[#0e1c14]/15 to-[#0e1c14]/50" />
   573→      </div>
   574→
   575→      {/* HUD */}
   576→      <header className="relative z-10 flex items-center gap-2 px-3 pt-3">
   577→        <IconButton img={A('icon-pause')} label="Pause" onClick={() => { pausedRef.current = true; setPaused(true) }} />
   578→        <div className="flex-1 min-w-0 text-center">
   579→          <p className="font-display font-extrabold text-[#fff4d4] ww-text-outline leading-tight truncate">{level.name}</p>
   580→          <p className="text-[10px] font-semibold tracking-[0.2em] uppercase text-[#f4e9c8]/75">Chapter {level.id}</p>
   581→        </div>
   582→        <div className="hud-pill rounded-2xl px-3 py-1 flex flex-col items-center leading-none" aria-label={`${movesLeft} moves left`}>
   583→          <span className="text-[9px] uppercase tracking-widest text-[#d9c79a] font-bold">Moves</span>
   584→          <span className={cn('font-display text-xl font-extrabold tabular-nums', movesLeft <= 3 && 'text-[#ff9d7a] animate-pulse')}>{movesLeft}</span>
   585→        </div>
   586→      </header>
   587→
   588→      {/* score + objectives */}
   589→      <section className="relative z-10 px-3 mt-2" aria-label="Score and objectives">
   590→        <div className="flex items-center gap-2">
   591→          <span className="hud-pill rounded-full px-2.5 py-1 text-sm font-extrabold tabular-nums" aria-live="polite">
   592→            {score.toLocaleString()}
   593→          </span>
   594→          <div className="relative flex-1 h-4 rounded-full bg-[#241a0e]/85 border border-[#a97b42]/70">
   595→            <div
   596→              className="absolute inset-y-0 left-0 rounded-full bg-gradient-to-b from-[#ffe28a] via-[#ffc94d] to-[#e8963c] transition-[width] duration-500"
   597→              style={{ width: `${fill * 100}%` }}
   598→            />
   599→            {[2, 3].map((sn) => {
   600→              const threshold = sn === 2 ? level.star2 : level.star3
   601→              const lit = score >= threshold
   602→              return (
   603→                <img
   604→                  key={sn}
   605→                  src={A('star-sparkle')}
   606→                  alt=""
   607→                  draggable={false}
   608→                  className={cn('absolute w-6 h-6 -top-1 object-contain transition', lit ? '' : 'grayscale opacity-45')}
   609→                  style={{ left: `calc(${(threshold / star3) * 100}% - 12px)` }}
   610→                />
   611→              )
   612→            })}
   613→          </div>
   614→        </div>
   615→        <div className="flex flex-wrap items-center justify-center gap-2 mt-2">
   616→          {obj.kind === 'collect'
   617→            ? (obj.collect ?? []).map((goal) => (
   618→                <GoalChip key={goal.type} type={goal.type} have={collected[goal.type] ?? 0} need={goal.count} />
   619→              ))
   620→            : (
   621→              <GoalChip type="star" have={Math.min(score, star3)} need={star3} />
   622→            )}
   623→        </div>
   624→      </section>
   625→
   626→      {/* board */}
   627→      <main ref={wrapRef} className="relative z-10 flex-1 min-h-0 flex items-center justify-center px-3 py-2">
   628→        <div
   629→          className="board-frame"
   630→          style={{ width: size.w + 12, height: size.h + 12, padding: 5, opacity: size.w > 0 ? 1 : 0 }}
   631→        >
   632→          <div
   633→            ref={boardRef}
   634→            role="application"
   635→            aria-label={`${level.rows} by ${level.cols} weaving board. Drag or tap charms to swap.`}
   636→            className="relative w-full h-full rounded-[0.7rem] overflow-hidden touch-none select-none"
   637→            onPointerDown={onPointerDown}
   638→            onPointerMove={onPointerMove}
   639→            onPointerUp={onPointerUp}
   640→            onPointerCancel={() => (dragRef.current = null)}
   641→            style={{
   642→              backgroundImage: 'repeating-conic-gradient(rgba(255,255,255,0.05) 0% 25%, transparent 0% 50%)',
   643→              backgroundSize: `${100 / level.cols}% ${100 / level.rows}%`,
   644→            }}
   645→          >
   646→            {grid.flatMap((row, r) =>
   647→              row.map((t, c) => {
   648→                if (!t) return null
   649→                return (
   650→                  <TileView
   651→                    key={t.id}
   652→                    tile={t}
   653→                    r={r}
   654→                    c={c}
   655→                    rows={level.rows}
   656→                    cols={level.cols}
   657→                    selected={!!selected && selected.r === r && selected.c === c}
   658→                    hinted={!!hint && hint.some((h) => h.r === r && h.c === c)}
   659→                  />
   660→                )
   661→              }),
   662→            )}
   663→
   664→            {/* floaters */}
   665→            {floaters.map((f) => (
   666→              <div
   667→                key={f.id}
   668→                className="absolute pointer-events-none z-30"
   669→                style={{ left: `${((f.c + 0.5) * 100) / level.cols}%`, top: `${((f.r + 0.5) * 100) / level.rows}%`, width: 0, height: 0 }}
   670→              >
   671→                <span className="anim-rise absolute font-display font-black text-lg text-[#ffe9a8] drop-shadow-[0_2px_2px_rgba(0,0,0,0.8)] whitespace-nowrap -translate-x-1/2">
   672→                  {f.text}
   673→                </span>
   674→              </div>
   675→            ))}
   676→
   677→            {/* combo banner */}
   678→            {combo && (
   679→              <div key={combo.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   680→                <span className="anim-combo font-display italic font-black text-3xl text-[#ffd76e] px-6 py-1.5 rounded-full border-2 border-[#f4c05a] bg-gradient-to-b from-[#2c4a6e]/95 to-[#16263e]/95 drop-shadow-[0_4px_10px_rgba(0,0,0,0.55)]">
   681→                  Combo ×{combo.n}!
   682→                </span>
   683→              </div>
   684→            )}
   685→
   686→            {/* objective complete flash */}
   687→            {goalFlash && (
   688→              <div key={goalFlash.id} className="absolute inset-0 z-40 flex items-center justify-center pointer-events-none">
   689→                <img src={A('banner-objective')} alt="Objective complete!" draggable={false} className="anim-combo w-48 drop-shadow-2xl" />
   690→              </div>
   691→            )}
   692→
   693→            {/* toast */}
   694→            {toast && (
   695→              <div className="absolute inset-x-0 bottom-3 z-40 flex justify-center pointer-events-none">
   696→                <span className="hud-pill rounded-full px-4 py-1.5 text-xs font-bold anim-fade-in">{toast}</span>
   697→              </div>
   698→            )}
   699→          </div>
   700→        </div>
   701→
   702→      </main>
   703→
   704→      <footer className="relative z-10 text-center pb-2">
   705→        <p className="text-[10px] text-[#f4e9c8]/60 ww-text-outline font-medium px-4 truncate">
   706→          {level.hint}
   707→        </p>
   708→      </footer>
   709→
   710→      {/* modals */}
   711→      {paused && status === 'playing' && (
   712→        <PauseModal
   713→          level={level}
   714→          onResume={() => {
   715→            pausedRef.current = false
   716→            setPaused(false)
   717→            resetHintTimer()
   718→          }}
   719→          onRestart={() => {
   720→            pausedRef.current = false
   721→            setPaused(false)
   722→            resetLevel()
   723→          }}
   724→          onOptions={() => setShowOptions(true)}
   725→          onQuit={() => {
   726→            pausedRef.current = false
   727→            setPaused(false)
   728→            onExitToAtlas()
   729→          }}
   730→        />
   731→      )}
   732→      {showOptions && (
   733→        <OptionsModal
   734→          sound={sound}
   735→          vibration={vibration}
   736→          onSound={onSound}
   737→          onVibration={onVibration}
   738→          playerName={playerName}
   739→          onRename={() => {}}
   740→          onClose={() => setShowOptions(false)}
   741→        />
   742→      )}
   743→      {result && won && (
   744→        <LevelCompleteModal
   745→          level={level}
   746→          result={result}
   747→          isLastLevel={level.id >= LEVELS.length}
   748→          onReplay={resetLevel}
   749→          onAtlas={onExitToAtlas}
   750→          onNext={() => onPlayLevel(Math.min(LEVELS.length, level.id + 1))}
   751→        />
   752→      )}
   753→      {result && !won && (
   754→        <LevelFailedModal level={level} score={result.score} onReplay={resetLevel} onAtlas={onExitToAtlas} />
   755→      )}
   756→    </div>
   757→  )
   758→}

   761→
   762→function TileView({
   763→  tile,
   764→  r,
   765→  c,
   766→  rows,
   767→  cols,
   768→  selected,
   769→  hinted,
   770→}: {
   771→  tile: Tile
   772→  r: number
   773→  c: number
   774→  rows: number
   775→  cols: number
   776→  selected: boolean
   777→  hinted: boolean
   778→}) {
   779→  const spawned = !!tile.spawned
   780→  const specialClass =
   781→    tile.special === 'lineH' || tile.special === 'lineV'
   782→      ? 'special-lineH'
   783→      : tile.special === 'bomb'
   784→        ? 'special-bomb'
   785→        : ''
   786→  return (
   787→    <div
   788→      className={cn('absolute left-0 top-0', tile.clearing && 'tile-clearing')}
   789→      style={{
   790→        width: `${100 / cols}%`,
   791→        height: `${100 / rows}%`,
   792→        transform: `translate(${c * 100}%, ${(spawned ? r - rows : r) * 100}%)`,
   793→        transition: spawned
   794→          ? `transform 430ms cubic-bezier(0.3, 0.85, 0.35, 1.12) ${c * 24}ms`
   795→          : 'transform 240ms cubic-bezier(0.3, 0.8, 0.35, 1.05)',
   796→        zIndex: selected ? 20 : 10,
   797→      }}
   798→    >
   799→      <div
   800→        className={cn(
   801→          'absolute inset-[7%] flex items-center justify-center',
   802→          selected && 'anim-tile-select z-20',
   803→          hinted && !selected && 'anim-wiggle',
   804→          specialClass,
   805→        )}
   806→        style={{ borderRadius: '26%' }}
   807→      >
   808→        <img
   809→          src={tile.special === 'prism' ? A('fx-rainbow') : TILE_IMG[tile.type]}
   810→          alt=""
   811→          draggable={false}
   812→          className={cn(
   813→            'w-full h-full object-contain drop-shadow-[0_3px_3px_rgba(0,0,0,0.4)] pointer-events-none',
   814→            tile.special === 'prism' && 'anim-prism scale-110',
   815→          )}
   816→        />
   817→        {tile.special === 'lineH' && (
   818→          <img src={A('icon-bolt')} alt="" draggable={false} className="absolute w-[46%] h-[46%] object-contain -rotate-90 drop-shadow" />
   819→        )}
   820→        {tile.special === 'lineV' && (
   821→          <img src={A('icon-bolt')} alt="" draggable={false} className="absolute w-[46%] h-[46%] object-contain drop-shadow" />
   822→        )}
   823→        {tile.special === 'bomb' && (
   824→          <span className="absolute inset-[18%] rounded-full bg-[radial-gradient(circle,rgba(255,150,80,0.55),transparent_70%)] animate-pulse" />
   825→        )}
   826→      </div>
   827→    </div>
   828→  )
   829→}
