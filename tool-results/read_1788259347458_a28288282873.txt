   421→   421→  /* ------- swap flow ------- */
   422→   422→  const attemptSwap = React.useCallback(
   423→   423→    async (a: Cell, b: Cell, opts?: { free?: boolean }): Promise<void> => {
   424→   424→      if (busyRef.current || pausedRef.current || statusRef.current !== 'playing') return
   425→   425→      if (!a || !b || typeof a.r !== 'number' || typeof b.r !== 'number') return
   426→   426→      if (!adjacent(a, b)) return
   427→   427→      lock(true)
   428→   428→      setSelectedBoth(null)
   429→   429→      setHint(null)
   430→   430→
   431→   431→      const g0 = cloneGrid(gridRef.current)
   432→   432→      const ta = g0[a.r]?.[a.c]
   433→   433→      const tb = g0[b.r]?.[b.c]
   434→   434→      if (!ta || !tb) {
   435→   435→        lock(false)
   436→   436→        return
   437→   437→      }
   438→   438→
   439→   439→      const swapped = cloneGrid(g0)
   440→   440→      swapped[a.r][a.c] = { ...tb }
   441→   441→      swapped[b.r][b.c] = { ...ta }
   442→   442→      setGridBoth(swapped)
   443→   443→      sfx.swap()
   444→   444→      await sleep(190)
   445→   445→
   446→   446→      const g1 = gridRef.current
   447→   447→      const prismInvolved = ta.special === 'prism' || tb.special === 'prism'
   448→   448→      const specialVsSpecial = !prismInvolved && ta.special !== 'none' && tb.special !== 'none'
   449→   449→      // g1 already contains the swapped tiles, so any shape present now was created by the swap
   450→   450→      const valid = prismInvolved || specialVsSpecial || findShapes(g1).length > 0
   451→   451→
   452→   452→      if (!valid) {
   453→   453→        setGridBoth(g0)
   454→   454→        sfx.invalid()
   455→   455→        await sleep(190)

   915→   915→            onExit()
   916→   916→          }}
   917→   917→        />
   918→   918→      )}
   919→   919→      {result && status === 'won' && (
   920→   920→        <FolioSealedModal
   921→   921→          level={level}

   927→   927→          onReplay={resetLevel}
   928→   928→          onExit={onExit}
   929→   929→        />
   930→   930→      )}
   931→   931→      {result && status === 'lost' && (
   932→   932→        <FolioLostModal level={level} isDaily={isDaily} score={result.score} onReplay={resetLevel} onExit={onExit} />
   933→   933→      )}
   934→   934→    </div>
   935→   935→  )
   936→   936→}
